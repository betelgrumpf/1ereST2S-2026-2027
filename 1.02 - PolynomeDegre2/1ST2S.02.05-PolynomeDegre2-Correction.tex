\begin{corrBox}
	\begin{center}
		\setlength{\extrarowheight}{3pt}
		\setlength{\tabcolsep}{0.3cm}
		\begin{tabular}{|c|c|c|c|c|c|}

			\cline{2-3}
			& oui & non & & & \\
			\hline $3x^2-2x+5$ & $\times$ & & $3$ & $-2$ & $5$ \\
			\hline $-x^2+4$ & $\times$ & & $-1$ & $0$ & $4$ \\
			\hline $5x-7$ & & $\times$ & & & \\
			\hline $x^3+2x^2$ & & $\times$ & & & \\
			\hline $(x-2)(3x+1)$ & $\times$ & & $3$ & $-5$ & $-2$ \\
			\hline $7-2x+0,5x^2$ & $\times$ & & $0,5$ & $-2$ & $7$ \\
			\hline $(x+1)^2-x^2$ & & $\times$ & & & \\
			\hline
		\end{tabular}
	\end{center}
	Détails :
	\begin{itemize}
		\item $5x-7$ est de degré 1 (pas de terme en $x^2$) et $x^3+2x^2$ est de degré 3.
		\item $(x-2)(3x+1)=3x^2+x-6x-2=3x^2-5x-2$.
		\item $(x+1)^2-x^2=x^2+2x+1-x^2=2x+1$ : les termes en $x^2$ s'annulent, c'est une fonction affine.
	\end{itemize}
\end{corrBox}

\exerciceDS{}

Pour chacune des fonctions suivantes définies sur \R, dire s'il s'agit d'une fonction polynôme de degré 2. Si oui, donner les valeurs de $a$, $b$ et $c$.

\begin{multicols}{2}
	\begin{enumerate}[label=\textbf{\alph*)}]
		\item $f(x)=-2x^2+x-3$
		\item $g(x)=4x^2$
		\item $h(x)=1-x^2+6x$
		\item $k(x)=3x(x-4)$
		\item $m(x)=(2x-1)(2x+1)-4x^2$
		\item $p(x)=(x-3)^2$
	\end{enumerate}
\end{multicols}

\medskip
\noindent\textbf{\color{red!75!black}Correction :}

\smallskip
\begin{corrBox}
	\textbf{a)} $f(x)=-2x^2+x-3$ : oui, avec $a=-2$, $b=1$ et $c=-3$.
\end{corrBox}

\medskip
\begin{corrBox}
	\textbf{b)} $g(x)=4x^2$ : oui, avec $a=4$, $b=0$ et $c=0$.
\end{corrBox}

\medskip
\begin{corrBox}
	\textbf{c)} $h(x)=1-x^2+6x=-x^2+6x+1$ : oui, avec $a=-1$, $b=6$ et $c=1$.
\end{corrBox}

\medskip
\begin{corrBox}
	\textbf{d)} $k(x)=3x(x-4)=3x^2-12x$ : oui, avec $a=3$, $b=-12$ et $c=0$.
\end{corrBox}

\medskip
\begin{corrBox}
	\textbf{e)} $m(x)=(2x-1)(2x+1)-4x^2=4x^2-1-4x^2=-1$ : non, $m$ est une fonction constante.
\end{corrBox}

\medskip
\begin{corrBox}
	\textbf{f)} $p(x)=(x-3)^2=x^2-6x+9$ : oui, avec $a=1$, $b=-6$ et $c=9$.
\end{corrBox}

\exerciceDS{}

\medskip
\noindent\textbf{\color{red!75!black}Correction :}

\smallskip
\begin{corrBox}
	\textbf{1)} La courbe coupe l'axe des abscisses aux points d'abscisses $-1$ et $3$.\\
	Les solutions de $f(x)=0$ sont $x=-1$ et $x=3$.
\end{corrBox}

\medskip
\begin{corrBox}
	\textbf{2)} On trace la droite horizontale d'équation $y=-3$ : elle coupe la courbe aux points d'abscisses $0$ et $2$.\\
	Les solutions de $f(x)=-3$ sont $x=0$ et $x=2$.
\end{corrBox}

\medskip
\begin{corrBox}
	\textbf{3)} L'axe de symétrie passe par le milieu des deux solutions : $(-1+3):2=1$.\\
	L'axe de symétrie est la droite verticale d'équation $x=1$.
\end{corrBox}

\medskip
\begin{corrBox}
	\textbf{4)} Le sommet est sur l'axe de symétrie, donc son abscisse est $1$.\\
	$f(1)=1^2-2\times1-3=1-2-3=-4$. Le sommet a pour coordonnées $S(1 ; -4)$.
\end{corrBox}

\medskip
\begin{corrBox}
	\textbf{5)} Les branches de la parabole sont tournées vers le haut ($a=1>0$) : $f$ admet un \textbf{minimum} égal à $-4$, atteint pour $x=1$.
\end{corrBox}

\medskip
\noindent\textbf{\color{red!75!black}Correction :}

\smallskip
\begin{corrBox}
	Pour une fonction de la forme $x\mapsto ax^2+c$, le sommet de la parabole est le point $(0 ; c)$.
	\begin{itemize}
		\item $\mathcal{C}_2$ et $\mathcal{C}_4$ ont pour sommet $(0 ; -1)$, donc $c=-1$ : il s'agit de $f$ et $g$.\\
		$\mathcal{C}_2$ est tournée vers le haut ($a>0$) : $\mathcal{C}_2$ représente $f$ ($a=2$).\\
		$\mathcal{C}_4$ est tournée vers le bas ($a<0$) : $\mathcal{C}_4$ représente $g$ ($a=-1$).
		\item $\mathcal{C}_1$ et $\mathcal{C}_3$ ont pour sommet $(0 ; 2)$, donc $c=2$ : il s'agit de $h$ et $k$.\\
		$\mathcal{C}_3$ est tournée vers le haut ($a>0$) : $\mathcal{C}_3$ représente $h$ ($a=\frac12$).\\
		$\mathcal{C}_1$ est tournée vers le bas ($a<0$) : $\mathcal{C}_1$ représente $k$ ($a=-2$).
	\end{itemize}
\end{corrBox}

\section*{\color{green!55!black}Forme factorisée et forme développée}
% =====================================================================

\exerciceDS{}

Pour chacune des fonctions suivantes définies sur \R :
\begin{itemize}
	\item développer l'expression et donner les valeurs de $a$, $b$ et $c$ ;
	\item donner les racines de la fonction polynôme.
\end{itemize}

\begin{multicols}{3}
	\begin{enumerate}[label=\textbf{\alph*)}]
		\item $f(x)=3(x-2)(x+4)$
		\item $g(x)=-(x+1)(x-5)$
		\item $h(x)=2(x-3)^2$
	\end{enumerate}
\end{multicols}

\medskip
\noindent\textbf{\color{red!75!black}Correction :}

\smallskip
\begin{corrBox}
	\textbf{a)} $f(x)=3(x-2)(x+4)=3\left(x^2+4x-2x-8\right)=3\left(x^2+2x-8\right)=3x^2+6x-24$.\\
	$a=3$, $b=6$, $c=-24$. Les racines de $f$ sont $x_1=2$ et $x_2=-4$.
\end{corrBox}

\medskip
\begin{corrBox}
	\textbf{b)} $g(x)=-(x+1)(x-5)=-\left(x^2-5x+x-5\right)=-\left(x^2-4x-5\right)=-x^2+4x+5$.\\
	$a=-1$, $b=4$, $c=5$. Les racines de $g$ sont $x_1=-1$ et $x_2=5$.
\end{corrBox}

\medskip
\begin{corrBox}
	\textbf{c)} $h(x)=2(x-3)^2=2(x-3)(x-3)=2\left(x^2-6x+9\right)=2x^2-12x+18$.\\
	$a=2$, $b=-12$, $c=18$. $h$ possède une racine double : $x_1=x_2=3$.
\end{corrBox}

\exerciceDS{}

\begin{enumerate}[label=\textbf{\arabic*)}]
	\item Développer les expressions suivantes :
	\begin{multicols}{2}
		\begin{enumerate}[label=\textbf{\alph*)}]
			\item $f(x)=-2(x+3)(x-1)$
			\item $g(x)=\dfrac{1}{2}(x-4)(x+2)$
		\end{enumerate}
	\end{multicols}
	\item Donner les solutions des équations $f(x)=0$ et $g(x)=0$.
	\item Écrire sous forme factorisée la fonction polynôme $p$ de degré 2 dont les racines sont $2$ et $-5$ et dont le coefficient $a$ est égal à $3$. Développer ensuite l'expression obtenue.
\end{enumerate}

\medskip
\noindent\textbf{\color{red!75!black}Correction :}

\smallskip
\begin{corrBox}
	\textbf{1) a)} $f(x)=-2(x+3)(x-1)=-2\left(x^2-x+3x-3\right)=-2\left(x^2+2x-3\right)=-2x^2-4x+6$.
\end{corrBox}

\medskip
\begin{corrBox}
	\textbf{1) b)} $g(x)=\dfrac12(x-4)(x+2)=\dfrac12\left(x^2+2x-4x-8\right)=\dfrac12\left(x^2-2x-8\right)=\dfrac12x^2-x-4$.
\end{corrBox}

\medskip
\begin{corrBox}
	\textbf{2)} D'après les formes factorisées :\\
	$f(x)=0 \iff x=-3$ ou $x=1$ \qquad et \qquad $g(x)=0 \iff x=4$ ou $x=-2$.
\end{corrBox}

\medskip
\begin{corrBox}
	\textbf{3)} $p(x)=a(x-x_1)(x-x_2)=3(x-2)(x-(-5))=3(x-2)(x+5)$.\\
	En développant : $p(x)=3\left(x^2+5x-2x-10\right)=3\left(x^2+3x-10\right)=3x^2+9x-30$.
\end{corrBox}

\section*{\color{green!55!black}Factoriser connaissant une racine}
% =====================================================================

\exerciceDS{}

On considère la fonction $f$ définie sur \R par $f(x)=3x^2-3x-6$.

\begin{enumerate}[label=\textbf{\arabic*)}]
	\item Vérifier que $2$ est une racine de la fonction polynôme $f$.
	\item En déduire que $f$ peut s'écrire sous la forme $f(x)=3(x-2)(x-x_2)$.
	\item En choisissant une valeur de $x$ judicieuse, déterminer $x_2$.
	\item Écrire $f$ sous forme factorisée et vérifier le résultat en développant.
\end{enumerate}

\medskip
\noindent\textbf{\color{red!75!black}Correction :}

\smallskip
\begin{corrBox}
	\textbf{1)} $f(2)=3\times2^2-3\times2-6=12-6-6=0$. Donc $2$ est une racine de $f$.
\end{corrBox}

\medskip
\begin{corrBox}
	\textbf{2)} $f(x)=3x^2-3x-6$, donc $a=3$. Comme $2$ est une racine, la forme factorisée s'écrit $f(x)=3(x-2)(x-x_2)$.
\end{corrBox}

\medskip
\begin{corrBox}
	\textbf{3)} On cherche $x_2$ tel que $3x^2-3x-6=3(x-2)(x-x_2)$.\\
	Pour $x=0$ : $-6=3\times(-2)\times(-x_2)$, soit $-6=6x_2$, donc $x_2=-1$.
\end{corrBox}

\medskip
\begin{corrBox}
	\textbf{4)} $f(x)=3(x-2)(x-(-1))=3(x-2)(x+1)$.\\
	Vérification : $3(x-2)(x+1)=3\left(x^2+x-2x-2\right)=3\left(x^2-x-2\right)=3x^2-3x-6$. \checkmark
\end{corrBox}

\exerciceDS{}

On considère la fonction $f$ définie sur \R par $f(x)=-2x^2+2x+12$.

\begin{enumerate}[label=\textbf{\arabic*)}]
	\item Calculer $f(-1)$, $f(1)$ et $f(3)$. Conjecturer une racine de la fonction polynôme $f$.
	\item Factoriser $f$.
	\item En déduire les solutions de l'équation $f(x)=0$.
\end{enumerate}

\medskip
\noindent\textbf{\color{red!75!black}Correction :}

\smallskip
\begin{corrBox}
	\textbf{1)} $f(-1)=-2\times(-1)^2+2\times(-1)+12=-2-2+12=8$.\\
	$f(1)=-2\times1^2+2\times1+12=-2+2+12=12$.\\
	$f(3)=-2\times3^2+2\times3+12=-18+6+12=0$.\\
	On conjecture (et on vient de vérifier) que $3$ est une racine de $f$.
\end{corrBox}

\medskip
\begin{corrBox}
	\textbf{2)} $a=-2$ et $3$ est une racine, donc $f(x)=-2(x-3)(x-x_2)$.\\
	Pour $x=0$ : $12=-2\times(-3)\times(-x_2)$, soit $12=-6x_2$, donc $x_2=-2$.\\
	Ainsi $f(x)=-2(x-3)(x+2)$.\\
	Vérification : $-2(x-3)(x+2)=-2\left(x^2+2x-3x-6\right)=-2\left(x^2-x-6\right)=-2x^2+2x+12$. \checkmark
\end{corrBox}

\medskip
\begin{corrBox}
	\textbf{3)} Les solutions de l'équation $f(x)=0$ sont les racines de $f$ : $x=3$ et $x=-2$.
\end{corrBox}

\exerciceDS{}

Une entreprise fabrique des objets. On admet que son bénéfice, en milliers d'euros, pour la fabrication de $x$ centaines d'objets (avec $x\in[0 ; 10]$), est donné par :
$$B(x)=-2(x-2)(x-8).$$

\begin{enumerate}[label=\textbf{\arabic*)}]
	\item Calculer $B(0)$ et $B(5)$. Interpréter ces résultats dans le contexte de l'exercice.
	\item Déterminer les racines de la fonction polynôme $B$.
	\item Dresser le tableau de signes de $B(x)$ sur l'intervalle $[0 ; 10]$.
	\item Pour quelles quantités d'objets fabriqués l'entreprise réalise-t-elle un bénéfice positif ?
	\item En utilisant la symétrie de la parabole, déterminer la quantité d'objets à fabriquer pour obtenir un bénéfice maximal. Quel est ce bénéfice maximal ?
\end{enumerate}

\medskip
\noindent\textbf{\color{red!75!black}Correction :}

\smallskip
\begin{corrBox}
	\textbf{1)} $B(0)=-2\times(0-2)\times(0-8)=-2\times(-2)\times(-8)=-32$.\\
	Sans fabriquer d'objets, l'entreprise réalise une perte de $32\,000$~\euro.\\[0.1cm]
	$B(5)=-2\times(5-2)\times(5-8)=-2\times3\times(-3)=18$.\\
	Pour $500$ objets fabriqués, le bénéfice est de $18\,000$~\euro.
\end{corrBox}

\medskip
\begin{corrBox}
	\textbf{2)} $x-2=0 \iff x=2$ \quad et \quad $x-8=0 \iff x=8$. Les racines de $B$ sont $2$ et $8$.
\end{corrBox}

\medskip
\begin{corrBox}
	\textbf{3)}
	\begin{center}
