\exerciceDS{}

Compléter le tableau suivant.

\begin{corrBox}
	\begin{center}
		\setlength{\extrarowheight}{3pt}
		\setlength{\tabcolsep}{0.3cm}
		\begin{tabular}{|c|c|c|c|c|c|}
			\hline
			\multirow{2}{*}{$f(x)=\ldots$} & \multicolumn{2}{c|}{Fonction polynôme de degré 2 ?} & \multirow{2}{*}{$a=\ldots$} & \multirow{2}{*}{$b=\ldots$} & \multirow{2}{*}{$c=\ldots$} \\
			\cline{2-3}
			& oui & non & & & \\
			\hline $4x^2+x-6$ & $\times$ & & $4$ & $1$ & $-6$ \\
			\hline $-3x^2+5$ & $\times$ & & $-3$ & $0$ & $5$ \\
			\hline $2x+9$ & & $\times$ & & & \\
			\hline $x^3-x$ & & $\times$ & & & \\
			\hline $(x+4)(2x-3)$ & $\times$ & & $2$ & $5$ & $-12$ \\
			\hline $3+x-2x^2$ & $\times$ & & $-2$ & $1$ & $3$ \\
			\hline $(x-3)^2-x^2$ & & $\times$ & & & \\
			\hline
		\end{tabular}
	\end{center}

	\textbf{Détails :}
	\begin{itemize}
		\item $2x+9$ est une fonction affine (degré 1) ; $x^3-x$ contient un terme en $x^3$ (degré 3).
		\item $(x+4)(2x-3)=2x^2-3x+8x-12=\mRouge{2x^2+5x-12}$.
		\item $3+x-2x^2=\mRouge{-2x^2+x+3}$ : attention à l'ordre des termes, $a$ est le coefficient de $x^2$.
		\item $(x-3)^2-x^2=x^2-6x+9-x^2=\mRouge{-6x+9}$ : les termes en $x^2$ s'annulent, c'est une fonction affine.
	\end{itemize}
\end{corrBox}

\exerciceDS{}

Pour chacune des fonctions suivantes définies sur \R, dire s'il s'agit d'une fonction polynôme de degré 2. Si oui, donner les valeurs de $a$, $b$ et $c$.

\begin{corrBox}
	\begin{enumerate}[label=\textbf{\alph*)}]
		\item $f(x)=3x^2-x+2$ : oui, avec $\mRouge{a=3}$, $\mRouge{b=-1}$ et $\mRouge{c=2}$.
		\item $g(x)=-5x^2$ : oui, avec $\mRouge{a=-5}$, $\mRouge{b=0}$ et $\mRouge{c=0}$.
		\item $h(x)=4-x+2x^2=2x^2-x+4$ : oui, avec $\mRouge{a=2}$, $\mRouge{b=-1}$ et $\mRouge{c=4}$.
		\item $k(x)=2x(x+3)=2x^2+6x$ : oui, avec $\mRouge{a=2}$, $\mRouge{b=6}$ et $\mRouge{c=0}$.
		\item $m(x)=(3x+1)(3x-1)-9x^2=9x^2-3x+3x-1-9x^2=-1$ : \mRougeWB{\textbf{non}}, $m$ est une fonction constante.
		\item $p(x)=(x+2)^2=x^2+4x+4$ : oui, avec $\mRouge{a=1}$, $\mRouge{b=4}$ et $\mRouge{c=4}$.
	\end{enumerate}
\end{corrBox}

\exerciceDS{}

\begin{minipage}[c]{0.5\linewidth}
	On a représenté ci-contre la fonction $f$ définie sur \R par
	$$f(x)=x^2+2x-3.$$
	\emph{En bleu : solutions de $f(x)=0$ ; en rouge : solutions de $f(x)=-3$ ; en orange : axe de symétrie.}
\end{minipage} \hfill
\begin{minipage}[c]{0.46\linewidth}
	\begin{center}
		\begin{tikzpicture}
			\begin{axis}[repere, width=7cm, height=8cm,
				xmin=-4.6, xmax=2.6, ymin=-4.6, ymax=5.6,
				xtick={-4,...,2}, ytick={-4,...,5}]
				\addplot[courbe, green!50!black, domain=-4.6:2.6] {x^2+2*x-3};
				\addplot[red, dashed, thick, domain=-4.6:2.6] {-3};
				\node[red, anchor=south west] at (axis cs:0.6,-3) {$y=-3$};
				\addplot[only marks, mark=*, red] coordinates {(-2,-3) (0,-3)};
				\addplot[only marks, mark=*, blue] coordinates {(-3,0) (1,0)};
				\draw[orange, thick, dashed] (axis cs:-1,-4.6) -- (axis cs:-1,5.6);
				\node[orange, anchor=west] at (axis cs:-1,4.8) {$x=-1$};
				\addplot[only marks, mark=*, violet] coordinates {(-1,-4)};
				\node[violet, anchor=north west] at (axis cs:-1,-4) {$S$};
			\end{axis}
		\end{tikzpicture}
	\end{center}
\end{minipage}

\begin{enumMet}
	\item Déterminer graphiquement les solutions de l'équation $f(x)=0$.

	\begin{corrBox}
		On lit les abscisses des points d'intersection de la courbe avec l'axe des abscisses : $\mRouge{S=\{-3\,;\,1\}}$.
	\end{corrBox}

	\item Déterminer graphiquement les solutions de l'équation $f(x)=-3$.

	\begin{corrBox}
		On lit les abscisses des points d'intersection de la courbe avec la droite d'équation $y=-3$ : $\mRouge{S=\{-2\,;\,0\}}$.
	\end{corrBox}

	\item En déduire l'équation de l'axe de symétrie de la courbe.

	\begin{corrBox}
		$-2$ et $0$ ont la même image, ils sont donc symétriques par rapport à l'axe de symétrie :
		\[x=\dfrac{-2+0}{2}=-1 \qquad \text{soit} \qquad \mRouge{x=-1}.\]
		\emph{(On retrouve le même résultat avec les racines : $\dfrac{-3+1}{2}=-1$.)}
	\end{corrBox}

	\item Calculer l'ordonnée du sommet $S$ de la parabole, puis donner ses coordonnées.

	\begin{corrBox}
		Le sommet est sur l'axe de symétrie, son abscisse est donc $-1$.

		$f(-1)=(-1)^2+2\times(-1)-3=1-2-3=-4$. \qquad Donc $\mRouge{S(-1\,;\,-4)}$.
	\end{corrBox}

	\item La fonction $f$ admet-elle un maximum ou un minimum ? Préciser sa valeur.

	\begin{corrBox}
		La parabole est tournée vers le haut ($a=1>0$) : $f$ admet un \mRougeWB{\textbf{minimum}} égal à $\mRouge{-4}$, atteint pour $x=-1$.
	\end{corrBox}
\end{enumMet}

\exerciceDS{}

\begin{minipage}[c]{0.40\linewidth}
	Associer chaque fonction à sa représentation graphique. Justifier.
	$$
	\begin{aligned}
		& f(x)=3x^{2}-2 \\
		& g(x)=-x^{2}+3 \\
		& h(x)=\frac{x^{2}}{2}+1 \\
		& k(x)=-2x^{2}-1
	\end{aligned}
	$$
\end{minipage} \hfill
\begin{minipage}[c]{0.56\linewidth}
	\begin{center}
		\begin{tikzpicture}
			\begin{axis}[repere, width=7.5cm, height=7.5cm,
				xmin=-3.6, xmax=3.6, ymin=-5.6, ymax=6.6,
				xtick={-3,...,3}, ytick={-5,...,6}]
				\addplot[courbe, red!80!white, line width=1.5pt, domain=-2.9:2.9] {-x^2+3};
				\addplot[courbe, blue!70!white, line width=1.5pt, domain=-1.5:1.5] {-2*x^2-1};
				\addplot[courbe, green!70!black, line width=1.5pt, domain=-1.65:1.65] {3*x^2-2};
				\addplot[courbe, orange, line width=1.5pt, domain=-3.3:3.3] {x^2/2+1};
				\addplot[only marks, mark=*, black] coordinates {(0,3) (0,-1) (0,-2) (0,1)};
				\node[red!80!white, anchor=west] at (axis cs:2.6,-3.6) {$\mathcal{C}_1$};
				\node[blue!70!white, anchor=west] at (axis cs:1.5,-5.2) {$\mathcal{C}_2$};
				\node[green!70!black, anchor=east] at (axis cs:-1.6,5.8) {$\mathcal{C}_3$};
				\node[orange, anchor=west] at (axis cs:2.6,3) {$\mathcal{C}_4$};
			\end{axis}
		\end{tikzpicture}
	\end{center}
\end{minipage}

\begin{corrBox}
	Pour une fonction $x\mapsto ax^2+c$ : si $a>0$ la parabole est tournée vers le haut, si $a<0$ elle est tournée vers le bas, et son sommet a pour coordonnées $(0\,;\,c)$.
	\begin{itemize}
		\item $f(x)=3x^2-2$ : $a=3>0$ (vers le haut), sommet $(0\,;\,-2)$ $\Rightarrow$ $f$ est représentée par $\mRouge{\mathcal{C}_3}$.
		\item $g(x)=-x^2+3$ : $a=-1<0$ (vers le bas), sommet $(0\,;\,3)$ $\Rightarrow$ $g$ est représentée par $\mRouge{\mathcal{C}_1}$.
		\item $h(x)=\dfrac{x^2}{2}+1$ : $a=\dfrac{1}{2}>0$ (vers le haut), sommet $(0\,;\,1)$ $\Rightarrow$ $h$ est représentée par $\mRouge{\mathcal{C}_4}$.
		\item $k(x)=-2x^2-1$ : $a=-2<0$ (vers le bas), sommet $(0\,;\,-1)$ $\Rightarrow$ $k$ est représentée par $\mRouge{\mathcal{C}_2}$.
	\end{itemize}
\end{corrBox}

\exerciceDS{}

On donne ci-dessous le tableau de variations d'une fonction polynôme du second degré $f$ définie sur \R par $f(x)=ax^2+bx+c$.

\begin{center}
	\begin{tikzpicture}
		\tikzset{node style/.append style = {fill=none}}
		\tkzTabInit[lgt = 1.6, espcl = 2.4, deltacl = 0.5]{$x$ / 1 , $f(x)$ / 2}{$-\infty$, $3$, $+\infty$}
		\tkzTabVar{+/, -/ $-4$, +/ }
	\end{tikzpicture}
\end{center}

\begin{enumMet}
	\item Le coefficient $a$ est-il positif ou négatif ? Justifier.

	\begin{corrBox}
		$f$ est d'abord décroissante puis croissante, donc $\mRouge{a>0}$.
	\end{corrBox}

	\item Donner les coordonnées du sommet $S$ de la parabole représentant $f$.

	\begin{corrBox}
		Le sommet correspond au changement de variation : $\mRouge{S(3\,;\,-4)}$.
	\end{corrBox}

	\item Donner l'équation de l'axe de symétrie de la parabole.

	\begin{corrBox}
		L'axe de symétrie passe par le sommet : $\mRouge{x=3}$.
	\end{corrBox}

	\item La fonction $f$ admet-elle un maximum ou un minimum sur \R ? Préciser sa valeur et pour quelle valeur de $x$ il est atteint.

	\begin{corrBox}
		$f$ admet un \mRougeWB{\textbf{minimum}} égal à $\mRouge{-4}$, atteint pour $\mRouge{x=3}$.
	\end{corrBox}

	\item Comparer, sans calcul, $f(5)$ et $f(7)$.

	\begin{corrBox}
		$5$ et $7$ appartiennent à l'intervalle $[3\,;\,+\infty[$ sur lequel $f$ est croissante. Comme $5<7$, on a $\mRouge{f(5)<f(7)}$.
	\end{corrBox}

	\item On sait que $f(0)=5$. En utilisant la symétrie de la parabole, donner la valeur de $f(6)$.

	\begin{corrBox}
		$\dfrac{0+6}{2}=3$ : les nombres $0$ et $6$ sont symétriques par rapport à l'axe d'équation $x=3$.

		Ils ont donc la même image : $\mRouge{f(6)=f(0)=5}$.
	\end{corrBox}
\end{enumMet}

\exerciceDS{}

On considère la fonction $f$ définie sur \R par $f(x)=-3x^2+12x-7$.

\begin{enumMet}
	\item Donner les valeurs de $a$, $b$ et $c$. En déduire le sens de variation de $f$ (d'abord croissante puis décroissante, ou l'inverse).

	\begin{corrBox}
		$\mRouge{a=-3}$, $\mRouge{b=12}$ et $\mRouge{c=-7}$.

		Comme $a=-3<0$, la fonction $f$ est \mRougeWB{\textbf{d'abord croissante puis décroissante}}.
	\end{corrBox}

	\item Calculer $f(1)$ et $f(3)$.

	\begin{corrBox}
		$f(1)=-3\times 1^2+12\times 1-7=-3+12-7=\mRouge{2}$

		$f(3)=-3\times 3^2+12\times 3-7=-27+36-7=\mRouge{2}$

		\emph{(Attention : $-3\times 3^2=-3\times 9=-27$ ; le carré ne porte que sur $3$.)}
	\end{corrBox}

	\item En déduire l'équation de l'axe de symétrie de la parabole représentant $f$.

	\begin{corrBox}
		$1$ et $3$ ont la même image, ils sont donc symétriques par rapport à l'axe de symétrie :
		\[x=\dfrac{1+3}{2}=2 \qquad \text{soit} \qquad \mRouge{x=2}.\]
	\end{corrBox}

	\item Calculer les coordonnées du sommet $S$ de la parabole.

	\begin{corrBox}
		Le sommet appartient à l'axe de symétrie, donc son abscisse est $2$.

		$f(2)=-3\times 2^2+12\times 2-7=-12+24-7=5$. \qquad Donc $\mRouge{S(2\,;\,5)}$.
	\end{corrBox}

	\item Dresser le tableau de variations de la fonction $f$.

	\begin{corrBox}
		\begin{center}
			\begin{tikzpicture}
				\tikzset{node style/.append style = {fill=none}}
				\tkzTabInit[lgt = 1.6, espcl = 2.4, deltacl = 0.5]{$x$ / 1 , $f(x)$ / 2}{$-\infty$, $2$, $+\infty$}
				\tkzTabVar{-/, +/ $5$, -/ }
			\end{tikzpicture}
		\end{center}
	\end{corrBox}
\end{enumMet}
