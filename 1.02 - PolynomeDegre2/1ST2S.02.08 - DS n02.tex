	On donne ci-dessous le tableau de variations d'une fonction polynôme du second degré $f$ définie sur $\mathbb{R}$ par $f(x)=ax^2+bx+c$.

	\begin{center}
		\begin{tikzpicture}
			\tikzset{node style/.append style = {fill=none}}
			\tkzTabInit[lgt = 1.6, espcl = 2.4, deltacl = 0.5]{$x$ / 1 , $f(x)$ / 2}{$-\infty$, $1$, $+\infty$}
			\tkzTabVar{+/, -/ $-3$, +/ }
		\end{tikzpicture}
	\end{center}

	\begin{enumMet}
		\item Le coefficient $a$ est-il positif ou négatif ? Justifier.
		\item Donner les coordonnées du sommet $S$ de la parabole représentant $f$, ainsi que l'équation de son axe de symétrie.
		\item La fonction $f$ admet-elle un maximum ou un minimum sur $\mathbb{R}$ ? Préciser sa valeur et pour quelle valeur de $x$ il est atteint.
		\item Comparer, sans calcul, $f(3)$ et $f(5)$. Justifier.
		\item \ding{46} On sait que $f(-1)=5$. En utilisant la symétrie de la parabole, donner la valeur de $f(3)$.
		\item Combien de solutions l'équation $f(x)=0$ admet-elle ? Justifier à l'aide du tableau.
	\end{enumMet}
