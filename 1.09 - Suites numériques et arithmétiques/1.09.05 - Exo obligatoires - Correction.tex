		On a donc $u_{20}=17+(20-25) \times 5=17+(-5) \times 5=-8$.		
	\end{corrBox}

		$$
		u_n=u_0+n \times 7
		$$
		On a donc $u_{20}=3+20 \times 7=3+20 \times 7=143$.			
	\end{corrBox}
	\item La suite $\left(u_n\right)$ est définie par $\left\{\begin{array}{l}u_1=-2 \\ u_{n+1}=u_n-4\end{array}\right.$ pour $n \in \mathbb{N}$.

Une famille décide d'épargner afin de pouvoir s'offrir un voyage en Égypte.

	\begin{corrBox}
		$s_1=500, s_2=600$ et $s_3=700$
		
		Pour tout entier $n \geqslant 1, s_{n+1}=s_n+100$.		
	\end{corrBox}
	\item En déduire l'expression de $s_n$ en fonction de l'entier naturel $n \geqslant 1$.
	\begin{corrBox}
		La suite est arithmétique de raison $r=100$ et de premier terme $u_1=500$ donc $u_n=500+(n-1) \times 100$ soit $u_n=100 n+400$.		
	\end{corrBox}
	\item A l'aide de la calculatrice, déterminer dans combien d'années la famille pourra partir en voyage sachant que le voyage coûte 4200 euros.
	\begin{corrBox}
		La somme totale épargnée est $S=s_1+s_2+\cdots+s_n=n \times \frac{s_1+s_n}{2}$. On doit résoudre $S \geqslant 4200$
		
		$$
		\begin{aligned}
			& \Longleftrightarrow n \times \frac{500+100 n+400}{2} \geq 4200 \\
			& \Longleftrightarrow 100 n^2+900 n \geq 8400 \\
			& \Longleftrightarrow 100 n^2+900 n-8400 \geq 0
		\end{aligned}
		$$
		On calcule le discriminant:
		
		$$
		\Delta=900^2+4 \times 8400=4170000
		$$
		
		
		Les racines sont:
		
		$$
		\begin{aligned}
			& n_1=\frac{-900-\sqrt{4170000}}{200}=\frac{-9-\sqrt{417}}{2}<0 \\
			& \text { et } n_2=\frac{-900+\sqrt{4170000}}{200}=\frac{-9+\sqrt{417}}{2} \approx 5,7 .
		\end{aligned}
		$$
		
		
		Le polynôme est du signe de $a=100$ donc positif à l'extérieur des racines.
		La famille pourra donc partir en voyage au bout de 6 ans.
	\end{corrBox}
\end{enumMet}

Pour stocker des photos numériques, on utilise un algorithme de compression. On estime qu'à chaque niveau de compression, la taille diminue de $21,4 \%$.
La taille initiale d'une photo est de 4 Mo . On pose $\mathrm{T}_0=4$ et, pour tout entier naturel non nul $n, \mathrm{~T}_n$ désigne la taille de cette photo après une compression de niveau $n$.

	\begin{corrBox}
		$$
		\begin{aligned}
			& T_1=4-0,214 \times 4=0,786 \times 4=3,144 \\
			& T_2=0,786 \times 3,144=2,471184
		\end{aligned}
		$$		
	\end{corrBox}

		La suite est donc géométrique de raison $q=0,786$ et de premier $T_0=4$.		
	\end{corrBox}
	\item Exprimer $\mathrm{T}_n$ en fonction de $n$.
	\begin{corrBox}
		Pour tout $n \in \mathbb{N}, T_n=4 \times 0,786^n$		
	\end{corrBox}

	\end{corrBox}
\end{enumMet}
