	\exercice{ $-$ Ex 50 p 172 du LS}

Simplifier au maximum les expressions suivantes.
\begin{enumMet}
	\item $A=e^2 \times e^{32} \times e^8$
	\item $\mathrm{B}=\dfrac{\mathrm{e}}{\mathrm{e}^5}$
	\item $\mathrm{C}=\dfrac{\left(\mathrm{e}^7\right)^3 \times \mathrm{e}^4}{\mathrm{e}^{-4}}$
\end{enumMet}

\exercice{ $-$ Ex 51 p 172 du LS}

$x$ est un réel quelconque. Simplifier au maximum les expressions suivantes.
\begin{enumMet}
	\item $a(x)=\mathrm{e}^{2 x} \times\left(\mathrm{e}^x\right)^2 \times \mathrm{e}^{-3 x}$
	\item $b(x)=\dfrac{\mathrm{e}^{x^2}}{\mathrm{e}^x}$
	\item $c(x)=\dfrac{\mathrm{e}^{x-1} \times \mathrm{e}^{4 x}}{\mathrm{e}^x}$
	\item $d(x)=\dfrac{\mathrm{e}^{-2 x}}{\mathrm{e}^{-3 x} \times \mathrm{e}^{x+1}}$
\end{enumMet}
\columnbreak
