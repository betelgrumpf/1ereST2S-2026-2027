	\exercice{ $-$ Ex 45 p 172 du LS}

$f$ est définie sur $\mathbb{R}$ par $f(x)=-2 \mathrm{e}^x$.

Vérifier que $f^{\prime}=f$ et calculer $f(0)$.

\begin{corrBox}
	Pour tout réel $x, f^{\prime}(x)=-2 \times 1 \times \mathrm{e}^x=f(x)$. De plus, $f(0)=-2 \mathrm{e}^0=-2$.	
\end{corrBox}

Déterminer une fonction $g$ définie et dérivable sur $\mathbb{R}$ telle que $g^{\prime}=g$ et $g(0)=\dfrac{3}{2}$.

\begin{corrBox}
	La fonction $g: x \mapsto \dfrac{3}{2} \mathrm{e}^x$ est définie et dérivable sur $\mathbb{R}$. On a de plus $g^{\prime}=g$ et $g(0)=\frac{3}{2} \mathrm{e}^0=\frac{3}{2}$.	
\end{corrBox}

\exercice{ $-$ Ex 51 p 172 du LS}

\begin{enumMet}

	\begin{corrBox}

	\begin{corrBox}
		$d(x)=\mathrm{e}^{-2 x-(-3 x+x+1)}=\mathrm{e}^{-1}$
	\end{corrBox}
\end{enumMet}

\begin{enumMet}
	\item $A=e^4\left(e^3+e^7\right)$
	\begin{corrBox}
		$A=\mathrm{e}^4 \times \mathrm{e}^3+\mathrm{e}^4 \times \mathrm{e}^7=\mathrm{e}^{4+3}+\mathrm{e}^{4+7}=\mathrm{e}^7+\mathrm{e}^{11}$		
	\end{corrBox}
	\item $B=\left(e^2+e^6\right)\left(e^3+e\right)$
	\begin{corrBox}
		$B=\mathrm{e}^2 \times \mathrm{e}^3+\mathrm{e}^2 \times \mathrm{e}^2+\mathrm{e}^6 \times \mathrm{e}^3+\mathrm{e}^6 \times \mathrm{e}=\mathrm{e}^5+\mathrm{e}^3+\mathrm{e}^9+\mathrm{e}^7$		
	\end{corrBox}
	\item $C=\left(e^8-e^2\right)\left(e^6+1\right)$
	\begin{corrBox}
		$C=\mathrm{e}^8 \times \mathrm{e}^6+\mathrm{e}^8-\mathrm{e}^2 \times \mathrm{e}^6-\mathrm{e}^2=\mathrm{e}^{14}+\mathrm{e}^8-\mathrm{e}^8+\mathrm{e}^2=\mathrm{e}^{14}+\mathrm{e}^2$		
	\end{corrBox}
	\item $\mathrm{D}=\left(\mathrm{e}^{-2}+\mathrm{e}^3\right)\left(\mathrm{e}^{-2}-\mathrm{e}^8\right)$
	\begin{corrBox}
		$D=\mathrm{e}^{-2} \times \mathrm{e}^{-2}-\mathrm{e}^{-2} \times \mathrm{e}^8+\mathrm{e}^3 \times \mathrm{e}^{-2}-\mathrm{e}^3 \times \mathrm{e}^8=\mathrm{e}^{-4}-\mathrm{e}^6+\mathrm{e}-\mathrm{e}^{11}$		
	\end{corrBox}
\end{enumMet}

\exercice{ $-$ Ex 30 p 171 du LS}

$x$ est un nombre réel. Simplifier les expressions.
\begin{enumMet}
	\item $\dfrac{\mathrm{e}^x \times\left(\mathrm{e}^x\right)^2}{\mathrm{e}^{2 x}}$
	\begin{corrBox}
		$\dfrac{\mathrm{e}^{x} \times\left(\mathrm{e}^{x}\right)^{2}}{\mathrm{e}^{2 x}}=\dfrac{\mathrm{e}^{x} \times \mathrm{e}^{2 x}}{\mathrm{e}^{2 x}}=\dfrac{\mathrm{e}^{x+2 x}}{\mathrm{e}^{2 x}}=\mathrm{e}^{3 x-2 x}=\mathrm{e}^{x}$		
	\end{corrBox}
	\item $\dfrac{\mathrm{e}^{x+4}}{\mathrm{e}^{4 x}}$
	\begin{corrBox}
		$\dfrac{\mathrm{e}^{\mathrm{x}+4}}{\mathrm{e}^{4 x}}=\mathrm{e}^{x+4-4 x}=\mathrm{e}^{-3 x+4}$		
	\end{corrBox}
	\item $\dfrac{1}{\mathrm{e}^{3-2 x}}$
	\begin{corrBox}
		$\dfrac{1}{\mathrm{e}^{3-2 x}}=\mathrm{e}^{-(3-2 x)}=\mathrm{e}^{2 x-3}$
	\end{corrBox}
\end{enumMet}

\begin{enumMet}
	\item $f(x)=3 \mathrm{e}^x$

	\end{corrBox}

	\item $h(x)=-\sqrt{2} \mathrm{e}^{-3 x}$

	\end{corrBox}
\end{enumMet}

\exercice{ $-$ Ex 32 p 171 du LS} 

Déterminer le signe des fonctions suivantes définies sur $\mathbb{R}$.
\begin{enumMet}
	\item $f(x)=\dfrac{1+\mathrm{e}^{4 x}}{x^2+2}$
	\begin{corrBox}
		Pour tout réel $x, \mathrm{e}^{4 x}>0$ donc le numérateur est strictement positif. De plus pour tout réel $x$, $x^{2}+2>0$. Donc, au final, $f$ est positive.		
	\end{corrBox}
	\item $g(x)=\dfrac{-9}{-2-\mathrm{e}^{-8 x}}$
	\begin{corrBox}
		Pour tout réel $x,-\mathrm{e}^{-8 x}<0,$ donc le dénominateur est strictement négatif. Le numérateur étant lui aussi négatif, la fonction $g$ est donc positive.		
	\end{corrBox}
\end{enumMet}

\exercice{ $-$ Ex 33 p 171 du LS}

\begin{enumMet}
	\item $f(x)=3 \mathrm{e}^x-5 x^2+2$
	\begin{corrBox}
		$f^{\prime}(x)=3 \mathrm{e}^x-5 \times 2 x+0=3 \mathrm{e}^x-10 x$.		
	\end{corrBox}

	\end{corrBox}

	\begin{corrBox}
		$f^{\prime}(x)=1 \times \mathrm{e}^x+x \times \mathrm{e}^x=(1+x) \mathrm{e}^x$
		 (dérivée d'un produit).		
	\end{corrBox}
\end{enumMet}

Calculer la dérivée de la fonction $f$ définie sur $\mathbb{R}$ et en déduire son tableau de variations.

	\end{corrBox}
	\item $f(x)=\mathrm{e}^{-x+4}$
	\begin{corrBox}

	\end{corrBox}
	\item $f(x)=5 \mathrm{e}^{x+6}$
	\begin{corrBox}
		$f^{\prime}(x)=5 \times 1 \times \mathrm{e}^{x+6}=5 \mathrm{e}^{x+6}, f$ est donc strictement croissante sur $\mathbb{R}$.		
	\end{corrBox}
\end{enumMet}

\exercice{ $-$ Ex 35 p 171 du LS}

Résoudre les équations suivantes dans $\mathbb{R}$.
\begin{enumMet}
	\item $\mathrm{e}^x=\mathrm{e}^{-2}$
	\begin{corrBox}
		$\mathrm{e}^x=\mathrm{e}^{-2} \Leftrightarrow x=-2$. Donc l'ensemble des solutions de cette équation est $S=\{-2\}$.		
	\end{corrBox}
	\item $\mathrm{e}^x=\mathrm{e}$
	\begin{corrBox}
		$\mathrm{e}^x=\mathrm{e} \Leftrightarrow \mathrm{e}^x=\mathrm{e}^1 \Leftrightarrow x=1$. Donc $S=\{1\}$.
	\end{corrBox}
	\item $\mathrm{e}^{x+2}=\mathrm{e}^3$
	\begin{corrBox}
		$\mathrm{e}^{x+2}=\mathrm{e}^3 \Leftrightarrow x+2=3 \Leftrightarrow x=1$. Donc $S=\{1\}$.
	\end{corrBox}
	\item $e^{2 x+1}=e$
	\begin{corrBox}
		$\mathrm{e}^{2 x+1}=\mathrm{e} \Leftrightarrow \mathrm{e}^{2 x+1}=\mathrm{e}^1 \Leftrightarrow 2 x+1=1 \Leftrightarrow x=0$. Donc $S=\{0\}$.
	\end{corrBox}
	\item $\mathrm{e}^x=1$
	\begin{corrBox}
		$\mathrm{e}^x=1 \Leftrightarrow \mathrm{e}^x=\mathrm{e}^0 \Leftrightarrow x=0$. Donc $S=\{0\}$.
	\end{corrBox}
	\item $\mathrm{e}^x+4=0$
	\begin{corrBox}
		$\mathrm{e}^x+4=0 \Leftrightarrow \mathrm{e}^x=-4$ ce qui est impossible car la fonction exponentielle est strictement positive sur $\mathbb{R}$. L'équation n'admet aucune solution : $S=\varnothing$.
	\end{corrBox}
	\item $\mathrm{e}^{x^2}=\mathrm{e}$
	\begin{corrBox}
		$\mathrm{e}^{x^2}=\mathrm{e} \Leftrightarrow \mathrm{e}^{x^2}=\mathrm{e}^1 \Leftrightarrow x^2=1 \Leftrightarrow x=1$ ou $x=-1$. Donc $S=\{-1 ; 1\}$.
	\end{corrBox}
	\item $\mathrm{e}^{x^2+1}=1$
	\begin{corrBox}
		$\mathrm{e}^{x^2+1}=1 \Leftrightarrow \mathrm{e}^{x^2+1}=\mathrm{e}^0 \Leftrightarrow x^2+1=0$ ce qui est impossible, donc $S=\varnothing$.
	\end{corrBox}
\end{enumMet}

\exercice{ $-$ Ex 36 p 171 du LS}

Résoudre les équations suivantes dans $\mathbb{R}$.
\begin{enumMet}
	\item $\mathrm{e}^{-x}=1$
	\begin{corrBox}
		$\mathrm{e}^{-x}=1 \Leftrightarrow \mathrm{e}^{-x}=\mathrm{e}^{0} \Leftrightarrow-x=0 \Leftrightarrow x=0$		
	\end{corrBox}
	\item $\mathrm{e}^{2 x-3}=\mathrm{e}$
	\begin{corrBox}
		$\mathrm{e}^{2 x-3}=\mathrm{e} \Leftrightarrow \mathrm{e}^{2 x-3}=\mathrm{e}^{1} \Leftrightarrow 2 x-3=1 \Leftrightarrow x=2$		
	\end{corrBox}
	\item $5 \mathrm{e}^{3 x+1}=5$
	\begin{corrBox}
		$5 \mathrm{e}^{3 x+1}=5 \Leftrightarrow \mathrm{e}^{3 x+1}=1 \Leftrightarrow \mathrm{e}^{3 x+1}=\mathrm{e}^{0} \Leftrightarrow 3 x+1=0 \Leftrightarrow x=-\dfrac{1}{3}$
	\end{corrBox}
	\item $-2 \mathrm{e}^{x^2}=3$
	\begin{corrBox}
		$-2 \mathrm{e}^{x^{2}}=3 \Leftrightarrow \mathrm{e}^{x^{2}}=-\dfrac{3}{2}$
		
		Ce qui est impossible car la fonction exponentielle est strictement positive sur $\mathbb{R}$. L'équation n'admet aucune solution : $S=\varnothing$.		
	\end{corrBox}
\end{enumMet}

\exercice{ $-$ Ex 37 p 171 du LS}

Résoudre les inéquations suivantes dans $\mathbb{R}$.
\begin{enumMet}
	\item $e^{2 x}>e^{-2}$
	\begin{corrBox}
		$\mathrm{e}^{2 x}>\mathrm{e}^{-2} \Leftrightarrow 2 x>-2 \Leftrightarrow x>-1$ donc l'ensemble des solutions est $\left.S=\right]-1 ;+\infty[$.		
	\end{corrBox}
	\item $\mathrm{e}^{-3 x}<\mathrm{e}$
	\begin{corrBox}
		$\mathrm{e}^{-3 x}<\mathrm{e} \Leftrightarrow \mathrm{e}^{-3 x}<\mathrm{e}^{1} \Leftrightarrow-3 x<1 \Leftrightarrow x>-\dfrac{1}{3}$\\
		Donc $S=\left]-\dfrac{1}{3} ;+\infty\right[$		
	\end{corrBox}
	\item $\mathrm{e}^{3 x-5} \geqslant \mathrm{e}^{-3}$
	\begin{corrBox}
		$\mathrm{e}^{3 x-5} \geqslant \mathrm{e}^{-3} \Leftrightarrow 3 x-5 \geqslant-3 \Leftrightarrow 3 x \geqslant 2 \Leftrightarrow x \geqslant \dfrac{2}{3}$\\
		Donc $S=\left[\dfrac{2}{3} ;+\infty\right[.$		
	\end{corrBox}
	\item $\mathrm{e}^{-2 x-1} \leqslant 1$
	\begin{corrBox}
		$\mathrm{e}^{-2 x-1} \leqslant 1 \Leftrightarrow \mathrm{e}^{-2 x-1} \leqslant \mathrm{e}^{0} \Leftrightarrow-2 x-1 \leqslant 0 \Leftrightarrow-2 x \leqslant 1 \Leftrightarrow x \geqslant-\dfrac{1}{2}$\\
		Donc $S=\left[\dfrac{1}{2} ;+\infty\right[$.		
	\end{corrBox}
\end{enumMet}
