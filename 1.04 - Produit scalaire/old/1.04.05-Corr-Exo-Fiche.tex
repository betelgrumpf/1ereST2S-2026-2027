\subsection*{Ex 47}
\begin{enumMet}
    \item
    \begin{enumMet}
        \item $AC=\sqrt{2^2+3^2}=\sqrt{13}$, $AB=\sqrt{1^2+6^2}=\sqrt{37}$ et $BC=\sqrt{4^2+4^2}=\sqrt{32}$
        \item $\vv{AB}\cdot\vv{AC}=\dfrac{1}{2}(AB^2+AC^2-BC^2)=\dfrac{1}{2}(37+13-32)=9$\\
        $\vv{AB}\cdot\vv{AC}=AB\times AC\times\cos(\widehat{BAC})$\\
        $\cos(\widehat{BAC})=\dfrac{\vv{AB}\cdot\vv{AC}}{AB\times AC}$\\
        $\cos(\widehat{BAC})=\dfrac{9}{\sqrt{37}\times \sqrt{13}}=\dfrac{9}{\sqrt{481}}\approx0,41$\\
        $\widehat{BAC}\approx65,8\degreees$
    \end{enumMet}
    \item 
    \begin{enumMet}
        \item $\vv{AB}\cdot\vv{AC}=AB\times AH$\\
        $9=\sqrt{37}\times AH$\\
        $AH=\dfrac{9}{\sqrt{37}}\approx1,5$
        On utilise Pythagore pour calculer CH (dans le triangle ACH)\\
        $CH^2=AC^2-AH^2$\\
        $CH^2=\dfrac{20\sqrt{37}}{37}\approx3,29$
        \item $A_{ABC}=\dfrac{AB\times CH}{2}=\dfrac{\sqrt{37}\times\dfrac{20\sqrt{37}}{37}}{2}=10$
    \end{enumMet}
\end{enumMet}

    $\|\vv{u}+\vv{v}\|^{2}=2\vv{u} \cdot \vv{v}+\|\vv{u}\|^{2}+\|\vv{v}\|^{2}=2\times(-4)+13+5=10$\\
    \grasInMet{2ème méthode:}\\
    $\Coord{u}{2}{-3}+\Coord{v}{1}{2}=\vv{u}+\vv{v} \CoordSV{3}{-1}$\\
    $\|\vv{u}+\vv{v}\|^{2}=3^2+(-1)^2=10$\\

    $\left\|\vv{u}-\vv{v}\right\|^2$\\

    $\|\vv{u}-\vv{v}\|^{2}=\|\vv{u}\|^{2}+\|\vv{v}\|^{2}-2\vv{u} \cdot \vv{v}=13+5-2\times(-4)=26$\\
    \grasInMet{2ème méthode:}\\
    $\Coord{u}{2}{-3}-\Coord{v}{1}{2}=\vv{u}+\vv{v} \CoordSV{1}{-5}$\\
    $\|\vv{u}+\vv{v}\|^{2}=1^2+(-5)^2=26$
