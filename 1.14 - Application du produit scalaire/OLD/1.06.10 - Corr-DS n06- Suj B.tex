\exerciceDS{ - Produit scalaire et repère - ( 4 points )}
Soit les vecteurs $\vv{u}\left(\begin{array}{l}3 \\ 2\end{array}\right), \vv{v}\left(\begin{array}{c}-4 \\ -1\end{array}\right)$ et $\vv{w}\left(\begin{array}{l}1 \\ 3\end{array}\right)$\\
Calculer les produits scalaires suivants.
\begin{enumMet}
    \item $\vv{u} \cdot \vv{v}$\\
    \begin{corrBox}
        $\vv{u} \cdot \vv{v}=3\times(-4)+2\times(-1)=-14$
    \end{corrBox}
    \item $\vv{w} \cdot \vv{v}$\\
    \begin{corrBox}
        $\vv{w} \cdot \vv{v}=1\times(-4)+3\times(-1)=-7$
    \end{corrBox}
%    \columnbreak
    \item \ding{46} $\vv{u} \cdot(\vv{v}+\vv{w})$\\
    \begin{corrBox}
        $\vv{u} \cdot(\vv{v}+\vv{w})= \vv{u} \cdot \vv{v}+\vv{u} \cdot \vv{w}$\\
        $\phantom{\vv{u} \cdot(\vv{v}+\vv{w})}=-14+3\times1+2\times3=-5$
    \end{corrBox}
    \item $(-2 \vv{u}) \cdot \vv{v}+3(\vv{v} \cdot \vv{w})$\\
    \begin{corrBox}
        $(-2 \vv{u}) \cdot \vv{v}+3(\vv{v} \cdot \vv{w})=-2\times(-14)+3\times(-7)=-49$
    \end{corrBox}
\end{enumMet}

\exerciceDS{ - Au golf - ( 2 points )}
\begin{minipage}[c]{.6\linewidth}
Sur un green de golf, trois positions de la balle sont repérées en $\mathrm{M},$ N et P sur le schéma ci-contre. Le triangle MNP est tel que MN=8, $\mathrm{MP}=5$ et $\widehat{M}=53^{\circ}$.\\
Calculer la valeur exacte de NP, puis son arrondi à 0,1 près.
\end{minipage}
\hfill
\begin{minipage}[c]{.35\linewidth}
\begin{center}
   \includegraphics[scale=0.8]{1.05.DS-IMG-01} 
\end{center}
\end{minipage}
\begin{corrBox}
    D'après le théorème d'Al-Kashi et avec les notations de la figure, on a :\\
    $NP^2=NM^2+MP^2-2\times NM\times MP \times \cos(\widehat{NMP})$\\
    $NP^2=64+25-80\cos(53^{\circ})$ d'où\\
    $NP=\sqrt{89-80\cos(53^{\circ})}\approx6,4$
\end{corrBox}

\exerciceDS{ - \ding{46} - produit scalaire dans un triangle - ( 2 points )}

Soit un triangle $\mathrm{ABC}$ de longueur $\mathrm{AB}=4, \mathrm{AC}=6$ et $\mathrm{BC}=9$
\begin{enumMet}
    \item \ding{46} Calculer le produit scalaire $\vv{B A} \cdot \vv{B C}$\\
    \begin{corrBox}
        $\vv{A B} \cdot \vv{BC}=\dfrac{1}{2}\left(A B^{2}+BC^{2}-AC^{2}\right)=\dfrac{1}{2}(4^2+9^2-6^2)=30,5$
    \end{corrBox}
    \item \ding{46} En déduire la mesure de l'angle $\widehat{ABC}$\\
    \begin{corrBox}
        $\vv{BA}\cdot\vv{BC}=BA\times BC\times\cos(\widehat{ABC})=4\times9\times\cos(\widehat{BAC})=36\cos(\widehat{BAC})$\\
        De plus, d'après la question 1 $\vv{A B} \cdot \vv{BC}=30,5$, donc\\
        $36\cos(\widehat{BAC})=30,5 \iff \cos(\widehat{BAC})=\dfrac{30,5}{36} \iff \widehat{BAC}=\cos^{-1}\left(\dfrac{30,5}{36}\right)\approx32,1^{\circ}$
    \end{corrBox}
\end{enumMet}

\exerciceDS{ - Angles dans un triangle - ( 3 points )}
Dans un repère orthonormé du plan, on considère les points $\mathrm{A}(-2 ;1), \mathrm{B}(3 ; 4)$ et $\mathrm{C}(2 ; 1)$.\\
Déterminer les angles du triangle $A B C$ à 0,1 degré près.\\
\begin{corrBox}
    On utilise le théorème d'AL-Kashi : Dans un triangle $\mathrm{ABC},$ on a, avec les notations de la figure :
    $$
    a^{2}=b^{2}+c^{2}-2 b c \cos \hat{A} \iff \hat{A}=\cos^{-1}\left(\dfrac{b^2+c^2-a^2}{2bc}\right)
    $$
    \begin{center}
        \includegraphics[scale=0.7]{1.04.IMG-03}
    \end{center}
    On commence donc par calculer les 3 longueurs:\\
    $\Coord{AB}{x_B-x_A}{y_B-y_A}=\Coord{AB}{3-(-2)}{4-1}=\Coord{AB}{5}{3} \Longrightarrow AB=\sqrt{5^2+3^2}=\sqrt{34}$\\
    $\Coord{AC}{x_C-x_A}{y_C-y_A}=\Coord{AC}{2-(-2)}{1-1}=\Coord{AC}{4}{0} \Longrightarrow AC=\sqrt{4^2+0^2}=4$\\
    $\Coord{BC}{x_C-x_B}{y_C-y_B}=\Coord{BC}{2-3}{1-4}=\Coord{BC}{-1}{-3} \Longrightarrow BC=\sqrt{(-1)^2+(-3)^2}=\sqrt{10}$\\
    On peut maintenant calculer les angles:\\
    $\widehat{BAC}=\cos^{-1}\left(\dfrac{AC^2+AB^2-BC^2}{2\times AC\times AB}\right)=\cos^{-1}\left(\dfrac{16+34-10}{2\times \sqrt{16}\times \sqrt{34}}\right)\approx31,0^{^\circ}$\\
    $\widehat{ABC}=\cos^{-1}\left(\dfrac{BC^2+AB^2-AC^2}{2\times BC\times AB}\right)=\cos^{-1}\left(\dfrac{34+10-16}{2\times \sqrt{34}\times \sqrt{10}}\right)\approx40,6^{^\circ}$\\
    $\widehat{ACB}=180-\widehat{BAC}-\widehat{ABC}=180-31,0-40.6=92,7^{^\circ}$
\end{corrBox}

\exerciceDS{ - Médiatrice dans un triangle - ( \textcolor{red}{7} points )}
On considère le triangle $\mathrm{ABC}$ tel que $\mathrm{A}(-2 ;1), \mathrm{B}(3 ; 4)$ et $\mathrm{C}(2 ; 1)$.
\begin{enumMet}
    \item Montrer que les coordonnées du milieu M du segment $[BC]$ sont $M(2,5;2,5)$\\
    \begin{corrBox}
        $x_M=\dfrac{x_B+x_C}{2}=\dfrac{3+2}{2}=\dfrac{5}{2}=2,5\quad \quad \quad \quad y_M=\dfrac{y_B+y_C}{2}=\dfrac{4+1}{2}=\dfrac{5}{2}=2,5$
    \end{corrBox}
    \item Déterminer une équation de la médiatrice  du segment $[BC]$ dans le triangle $\mathrm{ABC}$ (la médiatrice est la perpendiculaire à $[BC]$ passant par M).\\
    \begin{corrBox}
        La médiatrice du segment $[BC]$ est la droite passant par $M$ et perpendiculaire à $(BC)$. Le vecteur $\vv{BC}$ est donc un vecteur normal à la médiatrice du segment $[BC]$.\\
        Comme $\Coord{BC}{2-3}{1-4}=\Coord{BC}{-1}{-3}$, une équation de la hauteur est de la forme $-x-3y+c=0$.\\
        Or $M$ appartient à la médiatrice, donc $-x_{M}-3y_{M}+c=0$, donc $-1\times 2,5-3\times2,5+c=0$, donc $-2,5-7,5+c=0$, donc $c=10$.\\
        Une équation de la médiatrice du segment $[BC]$ est alors $-x-3y+10=0$.
    \end{corrBox}
    \item Montrer que les coordonnées du milieu N du segment $[AC]$ sont $N(0;1)$\\
    \begin{corrBox}
        $x_N=\dfrac{x_A+x_C}{2}=\dfrac{-2+2}{2}=\dfrac{0}{2}=0\quad \quad \quad \quad y_N=\dfrac{y_A+y_C}{2}=\dfrac{1+1}{2}=\dfrac{2}{2}=1$
    \end{corrBox}
    \item Déterminer une équation de la médiatrice  du segment $[AC]$ dans le triangle $\mathrm{ABC}$.\\
    \begin{corrBox}
        La médiatrice du segment $[AC]$ est la droite passant par $N$ et perpendiculaire à $(A C)$. Le vecteur $\vv{A C}$ est donc un vecteur normal à la médiatrice du segment $[AC]$.\\
        Comme $\Coord{AC}{2-(-2)}{1-1}=\Coord{AC}{4}{0}$, une équation de la médiatrice est de la forme $4x+c=0$.\\
        Or $N$ appartient à la médiatrice, donc $4x_{N}+c=0$, donc $4\times 0+c=0$, donc $c=0$.\\
        Une équation de la médiatrice du segment $[AC]$ est alors $4x=0$, ce qui équivaut à $x=0$. Il s'agit de l'axe des ordonnées.
    \end{corrBox}
    \item Déterminer les coordonnées du point d'intersection $\mathrm{O}$ des deux médiatrices.\\
    \begin{corrBox}
       Comme la 2ème médiatrice est l'axe des ordonnées, les 2 droites se coupent en $x=0$.\\
       On remplace $x$ par $0$ dans la 1ère équation et on trouve $y=\dfrac{10}{3}$.\\
       Le point O a pour coordonnées $O\left(0;\dfrac{10}{3}\right)$
    \end{corrBox}
    \item Montrer que les coordonnées du milieu P du segment $[AB]$ sont $P(0,5;2,5)$\\
    \begin{corrBox}
        $x_P=\dfrac{x_A+x_B}{2}=\dfrac{-2+3}{2}=\dfrac{1}{2}=0,5\quad \quad \quad \quad y_P=\dfrac{y_A+y_B}{2}=\dfrac{1+4}{2}=\dfrac{5}{2}=2,5$
    \end{corrBox}
    \item Calculer le produit scalaire $\vv{\mathrm{AB}} \cdot \vv{\mathrm{OP}}$. Quel résultat connu retrouve-t-on ?\\
    \begin{corrBox}
        $\Coord{OP}{0,5-0}{2,5-(-\dfrac{10}{3})}$, donc $\Coord{OP}{0,5}{-\dfrac{5}{6}}$ et $\Coord{AB}{3-(-2)}{4-1}$, donc $\Coord{AB}{5}{3}$.\\
        Alors $\vv{A B} \cdot \vv{OP}=5 \times 0,5+3 \times\left(-\dfrac{5}{6}\right)=\dfrac{5}{2}-\dfrac{5}{2}=0$.\\
        Les vecteurs $\vv{OP}$ et $\vv{AB}$ sont orthogonaux, donc $(OP)$ est la médiatrice du segment $[AB]$. On en déduit que les médiatrices sont concourantes en $O$.
    \end{corrBox}

\end{enumMet}

\exerciceDS{ - Équations cartésiennes - ( 3 points )}
Dans chaque cas, déterminer une équation cartésienne de la droite $d$ passant par le point $\mathrm{A}$ et perpendiculaire à la droite (BC).
\begin{enumMet}
    \item \ding{46} $\mathrm{A}(3 ; 4)$ et $\Coord{u}{-2}{4}$ un vecteur directeur de $(\mathrm{BC})$.\\
    \begin{corrBox}
        Comme $\Coord{u}{-2}{4}$ est un vecteur directeur de la droite $(B C)$, il est un vecteur normal à la droite $d$.\\
        Une équation cartésienne est de la forme $-2x+4 y+c=0$.\\
        Or, le point $A$ appartient à la droite $d$, donc $-2x_{A}+4y_{A}+c=0$, donc $-2 \times 3+4 \times 4+c=0$, donc $c=-10$.\\
        Une équation de la droite $d$ est donc $-2 x+4 y-10=0$ ou encore $-x+2y-5=0$.
    \end{corrBox}
    \item $\mathrm{A}(1 ; 2), \mathrm{B}(2 ;-1)$ et $\mathrm{C}(-3 ; 2)$.\\
    \begin{corrBox}
        Comme $\Coord{BC}{-5}{3}$  est un vecteur directeur de la droite $(B C)$, il est un vecteur normal à la droite $d$.\\
        Une équation cartésienne est de la forme $-5x+3y+c=0$.\\
        Or, le point $A$ appartient à la droite $d$, donc $-5x_{A}+3y_{A}+c=0$, donc $-5\times 1+3 \times 2+c=0$, donc $c=-1$.\\
        Une équation de la droite $d$ est donc $-5x+3 y-1=0$        
    \end{corrBox}
    \item $\mathrm{A}(-4 ; 5)$ et la droite $(\mathrm{BC})$ est parallèle à l'axe des abscisses.\\
    \begin{corrBox}
        La droite $(B C)$ étant horizontale, un vecteur directeur de cette droite et un vecteur normal à $d$ est $\Coord{u}{1}{0}$.\\
        Une équation cartésienne de $d$ est de la forme $x+c=0$.\\
        Or, le point $A$ appartient à la droite $d$, donc $x_{A}+c=0$, donc $-4+c=0$, donc $c=4$.\\
        Une équation de la droite $d$ est donc $x+4=0$.        
    \end{corrBox}
\end{enumMet}
