\exerciceDS{ - Reconnaître une suite arithmétique - ( 4,5 points )}
Reconnaître parmi les suites définies sur $\mathbb{N}$ ci-dessous celles qui sont arithmétiques et préciser alors leur premier terme et leur raison.
\begin{enumMet}
    \item $u_{n}=-2+3 n$ \quad\bareme{0,5}\\
    \begin{corrBox}
        On reconnaît une expression d'une suite arithmétique :\\
        $u_n=u_0+n\times r$ avec $u_0=-2$ et $r=3$
    \end{corrBox}
    \item $u_{n}=2 n^{2}+3$ \quad\bareme{0,5}\\
    \begin{corrBox}
        $u_0=3\quad\quad\quad u_1=5\quad\quad\quad u_2=11\quad\quad\quad\quad\quad\quad\quad\quad u_2-u_1=6\neq u_1-u_0=2$\\
        $u_n$ n'est donc pas une suite arithmétique.
    \end{corrBox}
    \item $u_{n}=\dfrac{n+5}{2}$ \quad\bareme{0,5}\\
    \begin{corrBox}
        On reconnaît une expression d'une suite arithmétique :\\
        $u_n=u_0+n\times r$ avec $u_0=\dfrac{5}{2}$ et $r=\dfrac{1}{2}$
    \end{corrBox}
    \item $u_{n}=\dfrac{1}{n+1}$ \quad\bareme{0,5}\\
    \begin{corrBox}
        $u_0=1\quad\quad\quad u_1=\dfrac{1}{2}\quad\quad\quad u_2=\dfrac{1}{3}\quad\quad\quad\quad\quad\quad\quad\quad u_2-u_1=-\dfrac{1}{6}\neq u_1-u_0=-\dfrac{1}{2}$\\
        $u_n$ n'est donc pas une suite arithmétique.
    \end{corrBox}
    \item $u_{0}=-1$ et $u_{n+1}=u_{n}+2 n$ \quad\bareme{0,5}\\
    \begin{corrBox}
        $u_{n+1}-u_n=u_n+2 n-u_n=2n$. La différence n'est pas constante.\\
        La suite $u_n$ n'est donc pas arithmétique. 
    \end{corrBox}
    \item $u_{0}=4$ et $u_{n+1}=1,5+u_{n}$ \quad\bareme{0,5}\\
    \begin{corrBox}
        $u_{n+1}-u_n=1,5+u_n-u_n=1,5=r$.\\
        La suite $u_n$ est une suite arithmétique de raison $r=1,5$ et de premier terme $u_0=4$. 
    \end{corrBox}
\end{enumMet}

\exerciceDS{ - monotonie d'une suite - ( 5 points )}
Étudier la monotonie des suites suivantes.
\begin{enumMet}
    \item pour tout entier naturel $n$, $u_{n}=\dfrac{1}{2 n+3}$.\\
    \begin{corrBox}
        $u_{n+1}-u_n=\dfrac{1}{2n+5}-\dfrac{1}{2n+3}=\dfrac{(2n+3)-(2n+5)}{(2n+3)(2n+5)}=\dfrac{-2}{(2n+3)(2n+5)}<0$\\
        La suite $u_n$ est décroissante. 
    \end{corrBox}
    \item Pour tout entier naturel $n, u_{n}=2\times3^{n}$.\\
    \begin{corrBox}
        $u_{n+1}-u_n=2\times3^{n+1}-2\times3^n=2\times3^n\times(3-1)=2\times2\times3^n>0$.\\
        La suite est croissante.
    \end{corrBox}
    \item Pour tout entier naturel $n, v_{n}=0,6^{n}$.\\
    \begin{corrBox}
        $v_{n+1}-v_n=0,6^{n+1}-0,6^n=0,6^n\times(0,6-1)=-0,4\times0,6^n<0$.\\
        La suite est décroissante.
    \end{corrBox}
    \item Pour tout entier naturel $n, w_{n}=(-5)^{n}$.\\
    \begin{corrBox}
        $w_{n+1}-w_n=(-5)^{n+1}-(-5)^n=(-5)^n\times(-5-1)=-6\times(-5)^n$.\\
        Si n est pair, $w_{n+1}-w_n<0$ et Si n est impair, $w_{n+1}-w_n>0$. La suite n'est donc pas monotone.
    \end{corrBox}
    \item \ding{46} Pour tout entier naturel $n, t_{n}=2 n+3$.\\
    \begin{corrBox}
        Pour tout entier $t_{n+1}=2(n+1)+3=2n+5$\\
        Pour tout entier $t_{n+1}-t_{n}=2n+5-(2n+3)=2>0$\\
        La suite $\left(t_{n}\right)$ est croissante.\\
    \end{corrBox}
\end{enumMet}

\exerciceDS{ - terme général d'une suite arithmétique - ( 4 points )}
Les suites suivantes sont arithmétiques de raison $r$. Exprimer $u_{n}$ en fonction de $n$, pour tout $n \in \mathbb{N}$.
\begin{enumMet}
    \item $u_{2}=5$ et $r=2$\\
    \begin{corrBox}
        $u_0=u_2-2r=5-2\times2=1\Longrightarrow u_n=u_0+nr \Longrightarrow u_n=1+2n$
    \end{corrBox}
    \item $u_{6}=8$ et $r=-\dfrac{1}{2}$\\
    \begin{corrBox}
        $u_0=u_6-6r=8+6\times\dfrac{1}{2}=11\Longrightarrow u_n=u_0+nr \Longrightarrow u_n=11-\dfrac{1}{2}n$
    \end{corrBox}
    \item \ding{46} $u_{3}=4$ et $r=8$\\
    \begin{corrBox}
        $u_0=u_3-3r=4-3\times8=-20\Longrightarrow u_n=u_0+nr \Longrightarrow u_n=-20+8n$
    \end{corrBox}
    \item $u_{8}=\dfrac{15}{2}$ et $r=-\dfrac{1}{4}$\\
    \begin{corrBox}
        $u_0=u_8-8r=\dfrac{15}{2}+8\times\dfrac{1}{4}=\dfrac{19}{2}\Longrightarrow u_n=u_0+nr \Longrightarrow u_n=\dfrac{19}{2}-\dfrac{1}{4}n$
    \end{corrBox}
\end{enumMet}

\exerciceDS{ - Calcul d'un terme - ( 4 points )}
Pour chacune des suites suivantes, calculer $u_{25}$.
\begin{enumMet}
    \item La suite $\left(u_{n}\right)$ est arithmétique de raison $r=2$ et telle que $u_{9}=12$.\\
    \begin{corrBox}
        $u_{25}=u_{9}+(25-9)\times r=12+(25-9)\times 2=44$
    \end{corrBox}
    \item La suite $\left(u_{n}\right)$ est arithmétique de raison $r=5$ et telle que $u_{30}=34$.\\
    \begin{corrBox}
        $u_{25}=u_{30}+(25-30)\times r=34+(25-30)\times 5=9$
    \end{corrBox}
    \item La suite $\left(u_{n}\right)$ est définie par $\left\{\begin{array}{l}u_{0}=8 \\ u_{n+1}=u_{n}+8\end{array}\right.$\\
    \begin{corrBox}
        $u_{25}=u_{0}+25\times r=8+25\times 8=208$
    \end{corrBox}
    \item \ding{46} La suite $\left(u_{n}\right)$ est définie par $\left\{\begin{array}{l}u_{1}=-5 \\ u_{n+1}=u_{n}-3\end{array}\right.$ pour $n \in \mathbb{N}$.\\
    \begin{corrBox}
        $u_{25}=u_{1}+(25-1)\times r=-5+(25-1)\times -3=-77$
    \end{corrBox}
\end{enumMet}
