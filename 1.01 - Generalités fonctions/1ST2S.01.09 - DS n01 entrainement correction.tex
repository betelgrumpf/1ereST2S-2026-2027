\exerciceDS{}

Soit la fonction $f$ définie sur $\mathbb{R}$ par $f(x)=-4x+7$.
\begin{enumMet}
	\item Calculer les images de $-3$ et de $2$ par $f$.

	\begin{corrBox}
		$f(-3)=-4\times(-3)+7=12+7=19$

		$f(2)=-4\times 2+7=-8+7=-1$

		L'image de $-3$ par $f$ est $\mRouge{19}$ et l'image de $2$ par $f$ est $\mRouge{-1}$.
	\end{corrBox}

	\item Déterminer un antécédent de $-5$ par $f$.

	\begin{corrBox}
		On résout l'équation $f(x)=-5$ :
		\[-4x+7=-5 \iff -4x=-5-7 \iff -4x=-12 \iff x=\dfrac{-12}{-4}=3\]
		Un antécédent de $-5$ par $f$ est $\mRouge{3}$. \quad (Vérification : $f(3)=-12+7=-5$.)
	\end{corrBox}
\end{enumMet}

	\begin{corrBox}
		Le point de $C_f$ d'abscisse $-5$ a pour ordonnée $5$.

		L'image de $-5$ par $f$ est $\mRouge{5}$ : $f(-5)=5$.
	\end{corrBox}

	\item Donner un antécédent de $-1$.

	\begin{corrBox}
		On cherche les abscisses des points de $C_f$ d'ordonnée $-1$ (intersection avec la droite d'équation $y=-1$).

		Un antécédent de $-1$ par $f$ est $\mRouge{-2}$.

		(Les autres antécédents de $-1$ sont environ $-2{,}7$ et $3{,}7$.)
	\end{corrBox}

	\begin{corrBox}
		On cherche les abscisses des points d'intersection de $C_f$ avec la droite d'équation $y=2$.

		$\mRouge{S\approx\left\{-4{,}3\ ;\ 1\ ;\ 2{,}3\right\}}$
	\end{corrBox}

	\item Résoudre graphiquement l'inéquation $f(x)< 1$.

	\begin{corrBox}
		On cherche les abscisses des points de $C_f$ situés strictement en dessous de la droite d'équation $y=1$.

		$\mRouge{S=\left]-4\,;\,0\right[\ \cup\ \left]3\,;\,4\right]}$
	\end{corrBox}
\end{enumMet}

\exerciceDS {}
Soit la fonction $f$ définie sur $\mathbb{R}$ par $f(x)=-3x+2$.
\begin{enumMet}
	\item Calculer le taux de variation de $f$ entre $-2$ et $1$.

	\begin{corrBox}
		$f(-2)=-3\times(-2)+2=6+2=8$ \quad et \quad $f(1)=-3\times 1+2=-3+2=-1$
		\[\dfrac{f(1)-f(-2)}{1-(-2)}=\dfrac{-1-8}{3}=\dfrac{-9}{3}=-3\]
		Le taux de variation de $f$ entre $-2$ et $1$ est $\mRouge{-3}$.
	\end{corrBox}

	\item Calculer le taux de variation de $f$ entre $2$ et $6$.

	\begin{corrBox}
		$f(2)=-3\times 2+2=-6+2=-4$ \quad et \quad $f(6)=-3\times 6+2=-18+2=-16$
		\[\dfrac{f(6)-f(2)}{6-2}=\dfrac{-16-(-4)}{4}=\dfrac{-12}{4}=-3\]
		Le taux de variation de $f$ entre $2$ et $6$ est $\mRouge{-3}$.

		\emph{Remarque : $f$ est une fonction affine de coefficient directeur $-3$, son taux de variation est donc toujours égal à $-3$.}
	\end{corrBox}
\end{enumMet}

\exerciceDS {}

Soit la fonction $f$ définie sur $\mathbb{R}$ par $f(x)=x^2+4x-1$.
\begin{enumMet}
	\item Déterminer le taux de variation de $f$ entre $1$ et $3$.

	\begin{corrBox}
		$f(1)=1^2+4\times 1-1=1+4-1=4$

		$f(3)=3^2+4\times 3-1=9+12-1=20$
		\[\dfrac{f(3)-f(1)}{3-1}=\dfrac{20-4}{2}=\dfrac{16}{2}=8\]
		Le taux de variation de $f$ entre $1$ et $3$ est $\mRouge{8}$.
	\end{corrBox}

	\item Déterminer le taux de variation de $f$ entre $-1$ et $1$.

	\begin{corrBox}
		$f(-1)=(-1)^2+4\times(-1)-1=1-4-1=-4$ \quad et \quad $f(1)=4$ (calculé précédemment)
		\[\dfrac{f(1)-f(-1)}{1-(-1)}=\dfrac{4-(-4)}{2}=\dfrac{8}{2}=4\]
		Le taux de variation de $f$ entre $-1$ et $1$ est $\mRouge{4}$.

		\emph{Remarque : $f$ n'est pas affine, son taux de variation dépend donc des valeurs choisies.}
	\end{corrBox}
\end{enumMet}

\exerciceDS{}

Soit la fonction $f$ définie sur $\mathbb{R}$ par $f(x)=-4x+5$ et deux réels $x_1$ et $x_2$.
\begin{enumMet}
	\item Calculer le taux de variation de $f$ entre $x_1$ et $x_2$. Simplifier au maximum l'expression.

	\begin{corrBox}
		Soient $x_1$ et $x_2$ deux réels distincts.
		\[\dfrac{f(x_2)-f(x_1)}{x_2-x_1}=\dfrac{(-4x_2+5)-(-4x_1+5)}{x_2-x_1}=\dfrac{-4x_2+5+4x_1-5}{x_2-x_1}=\dfrac{-4(x_2-x_1)}{x_2-x_1}=-4\]
		Le taux de variation de $f$ entre $x_1$ et $x_2$ est égal à $\mRouge{-4}$.
	\end{corrBox}

	\item En déduire les variations de $f$ sur $\mathbb{R}$.

	\begin{corrBox}
		Le taux de variation de $f$ entre deux réels quelconques est égal à $-4$, qui est strictement négatif.

		Donc $f$ est \mRougeWB{\textbf{strictement décroissante sur $\mathbb{R}$}}.
	\end{corrBox}
\end{enumMet}

\exerciceDS{}

Soit $f$ une fonction définie par sa représentation graphique dans un repère.

\begin{center}
	\psset{xunit=0.8cm,yunit=0.8cm,algebraic=true,dimen=middle,dotstyle=o,dotsize=5pt 0,linewidth=0.8pt,arrowsize=3pt 2,arrowinset=0.25}
	\begin{pspicture*}(-5.5,-5)(5.5,4)
		\psgrid[gridcolor=lightgray, subgriddiv=1, gridlabels=0pt](-6,-6)(6,6)
		\psaxes[mathLabel=false,xAxis=true,yAxis=true,Dx=1.,Dy=1.,ticksize=-3pt 0,subticks=1,labelFontSize=\footnotesize]{->}(0,0)(-5.5,-4.9)(5.5,4)
		\psplot[linewidth=1.2pt,linecolor=ForestGreen,plotpoints=600]{-6}{4}{0.5*(x+4)*(x-1)-1}
		% lectures graphiques
		\psline[linecolor=red,linestyle=dashed](-4,0)(-4,-1)(0,-1)
		\psline[linecolor=red,linestyle=dashed](-3,0)(-3,-3)(0,-3)
		\psline[linecolor=blue,linestyle=dashed](-1,0)(-1,-4)(0,-4)
		\psline[linecolor=blue,linestyle=dashed](2,0)(2,2)(0,2)
		\psline[linecolor=orange,linestyle=dashed](1,0)(1,-1)(0,-1)
		\psdots[dotstyle=*,linecolor=red](-4,-1)(-3,-3)
		\psdots[dotstyle=*,linecolor=blue](-1,-4)(2,2)
		\psdots[dotstyle=*,linecolor=orange](1,-1)
		\rput[B](2.4,0.6){\textcolor{ForestGreen}{$\bm{C_f}$}}
	\end{pspicture*}
\end{center}

Avec la précision permise par le graphique :

\begin{enumMet}
	\item Déterminer le taux de variation de $f$ entre $-4$ et $-3$.

	\begin{corrBox}
		Par lecture graphique : $f(-4)=-1$ et $f(-3)=-3$.
		\[\dfrac{f(-3)-f(-4)}{-3-(-4)}=\dfrac{-3-(-1)}{1}=\dfrac{-2}{1}=-2\]
		Le taux de variation de $f$ entre $-4$ et $-3$ est $\mRouge{-2}$.
	\end{corrBox}

	\item Déterminer le taux de variation de $f$ entre $-1$ et $2$.

	\begin{corrBox}
		Par lecture graphique : $f(-1)=-4$ et $f(2)=2$.
		\[\dfrac{f(2)-f(-1)}{2-(-1)}=\dfrac{2-(-4)}{3}=\dfrac{6}{3}=2\]
		Le taux de variation de $f$ entre $-1$ et $2$ est $\mRouge{2}$.
	\end{corrBox}

	\item Déterminer le taux de variation de $f$ entre $-4$ et $1$. Que remarque-t-on ?

	\begin{corrBox}
		Par lecture graphique : $f(-4)=-1$ et $f(1)=-1$.
		\[\dfrac{f(1)-f(-4)}{1-(-4)}=\dfrac{-1-(-1)}{5}=\dfrac{0}{5}=0\]
		Le taux de variation de $f$ entre $-4$ et $1$ est $\mRouge{0}$.

		On remarque que $f(-4)=f(1)$ : un taux de variation nul ne signifie pas que $f$ est constante entre $-4$ et $1$ (ici $f$ décroît puis croît).
	\end{corrBox}
\end{enumMet}
