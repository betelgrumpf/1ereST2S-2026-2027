	\exerciceDS{ $-$ (3 points)}
	
	Soit la fonction $f$ définie sur $\mathbb{R}$ par $f(x)=3 x-5$.
	\begin{enumMet}
		\item Calculer les images de $-2$ et de $5$ par $f$.
		
		\begin{corrBox}
			$f(-2)=3\times(-2)-5=-6-5=-11$ \quad \bareme{1}
			
			$f(5)=3\times 5-5=15-5=10$ \quad \bareme{1}
			
			L'image de $-2$ par $f$ est $\mRouge{-11}$ et l'image de $5$ par $f$ est $\mRouge{10}$.
		\end{corrBox}
		
		\item Déterminer un antécédent de $4$ par $f$. \quad \bareme{1}
		
		\begin{corrBox}
			On résout l'équation $f(x)=4$ :
			\[3x-5=4 \iff 3x=9 \iff x=\dfrac{9}{3}=3\]
			Un antécédent de $4$ par $f$ est $\mRouge{3}$. \quad (Vérification : $f(3)=9-5=4$.)
		\end{corrBox}
	\end{enumMet}

		\end{corrBox}

	\exerciceDS { $-$ (4 points)}
	Soit la fonction $f$ définie sur $\mathbb{R}$ par $f(x)=-2 x+1$.
	\begin{enumMet}
		\item Calculer le taux de variation de $f$ entre $-1$ et $2$ .
		
		\begin{corrBox}
			$f(-1)=-2\times(-1)+1=2+1=3$ \quad \bareme{0,5} \quad et \quad $f(2)=-2\times 2+1=-4+1=-3$ \quad \bareme{0,5}
			\[\dfrac{f(2)-f(-1)}{2-(-1)}=\dfrac{-3-3}{3}=\dfrac{-6}{3}=-2\]
			Le taux de variation de $f$ entre $-1$ et $2$ est $\mRouge{-2}$. \quad \bareme{1}
		\end{corrBox}
		
		\item \ding{46} Calculer le taux de variation de $f$ entre $3$ et $5$.
		
		\begin{corrBox}
			$f(3)=-2\times 3+1=-6+1=-5$ \quad \bareme{0,5} \quad et \quad $f(5)=-2\times 5+1=-10+1=-9$ \quad \bareme{0,5}
			\[\dfrac{f(5)-f(3)}{5-3}=\dfrac{-9-(-5)}{2}=\dfrac{-4}{2}=-2\]
			Le taux de variation de $f$ entre $3$ et $5$ est $\mRouge{-2}$. \quad \bareme{1}
			
			\emph{Remarque : $f$ est une fonction affine de coefficient directeur $-2$, son taux de variation est donc toujours égal à $-2$.}
		\end{corrBox}
	\end{enumMet}

	\exerciceDS { $-$ (3 points)}
	
	Soit la fonction $f$ définie sur $\mathbb{R}$ par $f(x)=x^2-2x+3$
	
	Déterminer le taux de variation de $f$ entre 2 et 4 .
	
	\begin{corrBox}
		$f(2)=2^2-2\times 2+3=4-4+3=3$ \quad \bareme{1}
		
		$f(4)=4^2-2\times 4+3=16-8+3=11$ \quad \bareme{1}
		\[\dfrac{f(4)-f(2)}{4-2}=\dfrac{11-3}{2}=\dfrac{8}{2}=4\]
		Le taux de variation de $f$ entre $2$ et $4$ est $\mRouge{4}$. \quad \bareme{1}
	\end{corrBox}

	\exerciceDS{ $-$ (3 points)}
	
	Soit la fonction $f$ définie sur $\mathbb{R}$ par $f(x)=3x-2$ et deux réels $x_1$ et $x_2$.
	\begin{enumMet}
		\item Calculer le taux de variation de $f$ entre $x_1$ et $x_2$. Simplifier au maximum l'expression. \quad \bareme{2}
		
		\begin{corrBox}
			Soient $x_1$ et $x_2$ deux réels distincts.
			\[\dfrac{f(x_2)-f(x_1)}{x_2-x_1}=\dfrac{(3x_2-2)-(3x_1-2)}{x_2-x_1}=\dfrac{3x_2-2-3x_1+2}{x_2-x_1}=\dfrac{3(x_2-x_1)}{x_2-x_1}=3\]
			Le taux de variation de $f$ entre $x_1$ et $x_2$ est égal à $\mRouge{3}$.
		\end{corrBox}
		
		\item En déduire les variations de $f$ sur $\mathbb{R}$.  \quad \bareme{1}
		
		\begin{corrBox}
			Le taux de variation de $f$ entre deux réels quelconques est égal à $3$, qui est strictement positif.
			
			Donc $f$ est \mRougeWB{\textbf{strictement croissante sur $\mathbb{R}$}}.
		\end{corrBox}
	\end{enumMet}

			\[\dfrac{f(-2)-f(-3)}{-2-(-3)}=\dfrac{-1-2}{1}=-3\]

			Le taux de variation de $f$ entre $1$ et $4$ est $\mRouge{2}$. \quad \bareme{0,5}
		\end{corrBox}
	\end{enumMet}
