\exerciceDS{}

Soit la fonction $f$ définie sur $\mathbb{R}$ par $f(x)=-4x+7$.
\begin{multicols}{2}
	\begin{enumMet}
		\item Calculer les images de $-3$ et de $2$ par $f$.
		\item Déterminer un antécédent de $-5$ par $f$.
	\end{enumMet}
\end{multicols}

\begin{enumMet}
	\item Calculer le taux de variation de $f$ entre $-2$ et $1$.

\exerciceDS{}

Soit la fonction $f$ définie sur $\mathbb{R}$ par $f(x)=-4x+5$ et deux réels $x_1$ et $x_2$.
\begin{enumMet}
	\item Calculer le taux de variation de $f$ entre $x_1$ et $x_2$. Simplifier au maximum l'expression.
	\item En déduire les variations de $f$ sur $\mathbb{R}$.
\end{enumMet}

\exerciceDS{}

\begin{enumMet}

	\item Déterminer le taux de variation de $f$ entre $-1$ et $2$.
	\item Déterminer le taux de variation de $f$ entre $-4$ et $1$. Que remarque-t-on ?
\end{enumMet}
