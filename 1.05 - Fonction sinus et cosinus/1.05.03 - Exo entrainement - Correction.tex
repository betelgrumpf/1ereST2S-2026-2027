\section*{Exercices facultatifs}

\exercice{}

On considère la fonction $f$ définie sur $\mathrm{R}$ par $f(x)=\cos (x)+x^2$.
\begin{enumMet}
	\item Montrer que $f$ est une fonction paire.
	\begin{corrBox}
		$f(-x)=\cos (-x)+(-x)^2=\cos(x)+x^2=f(x)$.\\
		On a bien $f(-x)=f(x)$ donc la fonction $f$ est paire.	
	\end{corrBox}
	\item Que peut-on en déduire sur la courbe représentative de $f$ ?
	\begin{corrBox}
		La courbe représentative de $f$ a une symétrie d'axe l'axe des ordonnées.	
	\end{corrBox}
\end{enumMet}

\vspace*{-0.3cm}
\exercice{}

On considère la fonction $g$ définie sur $\mathbb{R}$ par $g(x)=x+\sin (x)$.
\begin{enumMet}
	\item Montrer que $g$ est une fonction impaire.
	\begin{corrBox}
		$g(-x)=-x+\sin (-x)=-x-\sin(x)=-g(x)$.\\
		On a bien $g(-x)=-g(x)$ donc la fonction $g$ est impaire.	
	\end{corrBox}
	\item Que peut-on en déduire sur la courbe représentative de $g$ ?
	\begin{corrBox}
		La courbe représentative de $g$ a une symétrie centrale avec l'origine du repère comme centre de symétrie.	
	\end{corrBox}
\end{enumMet}

\exercice{}

En exprimant, pour tout réel $x, f(-x)$ à l'aide de $f(x)$, dire si les fonctions définies sur $\mathbb{R}$ ci-dessous sont paires ou impaires.
%\begin{multicols}{2}
\begin{enumMet}
	\item $f: x \mapsto x \times \sin (x)$
	\begin{corrBox}
		$f(-x)=-x\times \sin(-x)=-x\times (-\sin(x))=x\times \sin (x)=f(x)$\\
		La fonction est paire.	
	\end{corrBox}
	\item $f: x \mapsto x \times \cos (x)$
	\begin{corrBox}
		$f(-x)=-x\times \cos(-x)=-x\times (\cos(x))=-f(x)$\\
		La fonction est impaire.	
	\end{corrBox}
	\item $f: x \mapsto (\sin (x))^2$
	\begin{corrBox}
		$f(-x)=(\sin(-x))^2=(-\sin(x))^2=(\sin(x))^2=f(x)$\\
		La fonction est paire.	
	\end{corrBox}
	\item $f: x \mapsto \dfrac{x^2}{2+\cos (x)}$
	\begin{corrBox}
		$f(-x)=\dfrac{(-x)^2}{2+\cos (-x)}=\dfrac{x^2}{2+\cos (x)}=f(x)$\\
		La fonction est paire.	
	\end{corrBox}
\end{enumMet}

\exercice{}

Donner à chaque courbe représentative la fonction qui lui correspond. Justifier la réponse.
\begin{multicols}{2}
	\begin{enumMet}
		\item $f: x \mapsto 2 \sin (x)$
		\item $g: x \mapsto \cos (2 x)$
	\end{enumMet}
\end{multicols}
\begin{center}
	\includegraphics[scale=1]{1.03.02-Ex45LLS}
\end{center}
\begin{corrBox}
On a $f(0)=2\sin(0)=0$ et $f\left(\dfrac{\pi}{2}\right)=2\sin\left(\dfrac{\pi}{2}\right)=2$\\
La courbe qui représente la fonction $f$ passe par le point de coordonnée $(0;0)$ et $\left(\dfrac{\pi}{2};2\right)$. Il s'agit de la courbe rouge.\\	
De même, on a $g(0)=\cos(2\times 0)=1$ et $g\left(\dfrac{\pi}{2}\right)=\cos\left(2\times \dfrac{\pi}{2}\right)=\cos(\pi)=-1$\\
La courbe qui représente la fonction $f$ passe par le point de coordonnée $(0;0)$ et $\left(\dfrac{\pi}{2};-1\right)$. Il s'agit de la courbe verte.	
\end{corrBox}
