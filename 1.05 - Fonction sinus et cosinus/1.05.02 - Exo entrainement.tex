résoudre l'équation donnée dans l'intervalle I.

\begin{multicols}{2}
	\begin{enumMet}
	\item $\cos (x)=\frac{\sqrt{3}}{2} \quad$ avec $\mathrm{I}=\mathbb{R}$.
	\item $\sin (x)=-\frac{\sqrt{2}}{2} \quad$ avec $I=\mathbb{R}$.
	\item $\sin (x)=\frac{\sqrt{3}}{2} \quad$ avec $I=\mathbb{R}$.
	\item $\cos (x)=\frac{1}{2} \quad$ avec $I=\mathbb{R}$.
	\item $\cos (x)=-\frac{\sqrt{2}}{2} \quad$ avec $\mathrm{I}=[0 ; 2 \pi[$.
	\item $\sin (x)=\frac{1}{2} \quad$ avec $\left.\left.\mathrm{I}=\right]-\pi ; \pi\right]$.
	\item $\cos (x)=0 \quad$ avec $\mathrm{I}=]-\pi ; \pi]$.
	\item $\cos (x)=-\frac{1}{2} \quad$ avec $\mathrm{I}=\left]-\pi ; \pi\right]$.
	\item $\sin (x)=0 \quad$ avec $I=]-\pi ; \pi]$.
	\item $\cos (x)=-\frac{\sqrt{3}}{2} \quad$ avec $\mathrm{I}=[0 ; 4 \pi[$.
\end{enumMet}
\end{multicols}
