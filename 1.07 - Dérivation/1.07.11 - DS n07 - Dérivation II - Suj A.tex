\exerciceDS{ - fonction dérivée - ( 9 points )}

Calculer la dérivé des fonctions suivantes
\begin{enumMet}
	\item $f(x)=\dfrac{3}{x}+4 \sqrt{x}$
	\item \ding{46} $g(x)=\dfrac{4}{3} x^{3}-0,5 x^{2}-\dfrac{2}{x}$
	\item $h(x)=2 x^{2} \sqrt{x}$
	\item $k(x)=\dfrac{3 x-4}{x^{2}-5}$
	\item $p(x)=\dfrac{4}{3x^2-4x+3}-3x^2$
	\item $q(x)=\dfrac{5x^2-2x+1}{3x^2-5x+5}$
\end{enumMet}

\exerciceDS{ - problème de dérivation - ( 7 points )}
Soit $f$ la fonction définie sur $\left] \dfrac{2}{3} ;+\infty\right[$ par: $f(x)=\dfrac{3 x^{2}-11 x+11}{-3 x+2}$
\begin{enumMet}
	\item Déterminer l'expression de $f^{\prime}$
	\item Vérifier que pour tout réel $x$ tel que $x>\dfrac{2}{3}$\\ $f(x)=-x+3+\dfrac{5}{-3 x+2}$
	\item En déduire une autre expression de $f^{\prime}$
	\item Vérifier la cohérence des résultats obtenus au 1) et au 3).
\end{enumMet}
