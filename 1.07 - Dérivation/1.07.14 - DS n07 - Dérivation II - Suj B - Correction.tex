\exerciceDS{ - problème de dérivation - ( 7 points )}
Soit $f$ la fonction définie sur $\left] \dfrac{4}{3} ;+\infty\right[$ par: $f(x)=\dfrac{3 x^{2}-10x+13}{-3 x+4}$
\begin{enumMet}
	\item Déterminer l'expression de $f^{\prime}$\quad \bareme{2}
	\begin{corrBox}
		$f^{\prime}(x)=\dfrac{(6 x-10)(-3 x+4)+3\left(3 x^{2}-10 x+13\right)}{(-3 x+4)^{2}}=\dfrac{-9 x^{2}+24 x-1}{(-3 x+4)^{2}}$
	\end{corrBox}
	\item Vérifier que pour tout réel $x$ tel que $x>\dfrac{4}{3}$\\ $f(x)=-x+2+\dfrac{5}{-3 x+4}$\quad \bareme{2}
	\begin{corrBox}
		$f(x)=-x+2+\dfrac{5}{-3 x+4}=\dfrac{(-x+2)(-3 x+4)+5}{-3 x+4}=\dfrac{3 x^{2}-10 x+13}{-3 x+4}$
	\end{corrBox}
	\item En déduire une autre expression de $f^{\prime}$\quad \bareme{2}
	\begin{corrBox}
		$f^{\prime}(x)=-1-\dfrac{5 \times(-3)}{(-3 x+4)^{2}}=\dfrac{-(-3 x+4)^{2}+15}{(-3 x+4)^{2}}=\dfrac{-9 x^{2}+24 x-1}{(-3 x+4)^{2}}$
	\end{corrBox}
	\item Vérifier la cohérence des résultats obtenus au 1) et au 3).\quad \bareme{1}
	\begin{corrBox}
		Les résultats des 2 réponses sont identiques.
	\end{corrBox}
\end{enumMet}
