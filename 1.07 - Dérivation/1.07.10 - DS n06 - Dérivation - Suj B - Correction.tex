\exerciceDS{ - dérivabilité - ( 3 points )}

Soit $f$ la fonction définie sur \R par $f(x)=x^3+2 x^2-5 x-5$\\
\ding{46} Montrer que la fonction f est dérivable pour $x=2$ et calculer $f'(2)$
\begin{corrBox}
	$f(2)=2^3+2\times 2^2-5\times 2-5=1$ et\\
	$f(2+h)=(2+h)^3+2(2+h)^2-5(2+h)-5$\\
	$\phantom{f(2+h)}=2^3+3\times 2^2\times h+3\times 2\times h^2+h^3+2(4+4h+h^2)-10-5h-5$\\
	$\phantom{f(2+h)}=h^3+8h^2+15h+1$\\
	$\lim\limits_{h \rightarrow 0} \dfrac{f(2+h)-f(2)}{h}=\lim\limits_{h \rightarrow 0} \dfrac{h^3+8h^2+15h+1-1}{h}=\lim\limits_{h \rightarrow 0} \dfrac{h^3+8h^2+15h}{h}=\lim\limits_{h \rightarrow 0} h^2+8h+15=15$\\
	$15$ est un nombre réel donc f est dérivable pour $\mathrm{x}=2$ et $f^{\prime}(2)=15$.
\end{corrBox}

\exerciceDS{ - dérivabilité - ( 3 points )}
Soit f la fonction définie sur $] 1 ;+\infty[$ par $f(x)=\dfrac{1}{x-1}$.\\
Montrer que la fonction f est dérivable pour $x=3$ et calculer $f^{\prime}(3)$
\begin{corrBox}
	$f(3+h)-f(3)=\dfrac{1}{3+h-1}-\dfrac{1}{3-1}=\dfrac{1}{h+2}-\dfrac{1}{2}=\dfrac{2}{2(h+2)}-\dfrac{h+2}{2(h+2)}=\dfrac{-h}{2h+4}$\\
	$\lim\limits_{h \rightarrow 0} \dfrac{f(3+h)-f(3)}{h}=\lim\limits_{h \rightarrow 0} \dfrac{\dfrac{-h}{2h+4}}{h}=\lim\limits_{h \rightarrow 0} \dfrac{-1}{2h+4}=-\dfrac{1}{4}$\\
	$-\dfrac{1}{4}$ est un nombre réel donc f est dérivable pour $\mathrm{x}=3$ et $f^{\prime}(3)=-\dfrac{1}{4}$.
\end{corrBox}

	\item Calculs concernant la droite $D_1$
	\begin{enumMet}
		\item Calculer $f(-1)$ \quad \bareme{0,5}
		\begin{corrBox}
			$f(-1)=-(-1)^{2}+2\times (-1)+1=-2$	
		\end{corrBox}
		\item Calculer $f'(-1)$ \quad \bareme{2}
		\begin{corrBox}
			$f(-1+h)=-(-1+h)^{2}+2\times (-1+h)+1=-h^2+4h-2$\\
			$f'(-1)=\lim\limits_{h \rightarrow 0} \dfrac{f(-1+h)-f(-1)}{h}=\lim\limits_{h \rightarrow 0} \dfrac{-h^2+4h-2-(-2)}{h}=\lim\limits_{h \rightarrow 0} \dfrac{-h^2+4h}{h}=\lim\limits_{h \rightarrow 0} -h+4$\\
			$f'(-1)=4$
		\end{corrBox}
		\item A l'aide des 2 résultats précédents et d'un calcul, retrouver l'équation réduite de la droite $D_1$. \quad \bareme{1} 
		\begin{corrBox}
			$y=f'(a)(x-a)+f(a)$ avec $a=-1$\\
			$y=f'(-1)(x-(-1))+f(1)$\\
			$y=4(x+1)-2$ d'ou $y=4x+2$	
		\end{corrBox} 
	\end{enumMet}
	\item \ding{46} Calculs concernant la droite $D_3$

		\end{corrBox}
		\item \ding{46} Calculer $f'(2)$ \quad \bareme{2}
		\begin{corrBox}
			$f(2+h)=-(2+h)^{2}+2\times (2+h)+1=-h^2-2h+1$\\
			$f'(2)=\lim\limits_{h \rightarrow 0} \dfrac{f(2+h)-f(2)}{h}=\lim\limits_{h \rightarrow 0} \dfrac{-h^2-2h+1-1}{h}=\lim\limits_{h \rightarrow 0} \dfrac{-h^2-2h}{h}=\lim\limits_{h \rightarrow 0} -h-2$\\

		\end{corrBox}

		\end{corrBox}
	\end{enumMet}
