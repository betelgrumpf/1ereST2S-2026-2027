\exerciceDS{ - dérivabilité et graphique - ( 11,5 points )}

\begin{enumMet}
	\item A l'aide du graphique, trouver les équations réduites des droites $D_1$, $D_2$ et $D_3$. %\quad \bareme{3}
	\item En déduire $f'(-1)$, $f'(1)$ et $f'(2)$ %\quad \bareme{1,5}
	\item Calculs concernant la droite $D_1$
	\begin{enumMet}
		\item Calculer $f(-1)$ %\quad \bareme{0,5}
		\item Calculer $f'(-1)$ %\quad \bareme{2}
		\item A l'aide des 2 résultats précédents et d'un calcul, retrouver l'équation réduite de la droite $D_1$. \quad %\bareme{1}  
	\end{enumMet}
	\item \ding{46} Calculs concernant la droite $D_3$

	\end{enumMet}
