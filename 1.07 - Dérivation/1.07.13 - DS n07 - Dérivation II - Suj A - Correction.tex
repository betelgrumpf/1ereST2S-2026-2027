\exerciceDS{ - problème de dérivation - ( 7 points )}
Soit $f$ la fonction définie sur $\left] \dfrac{2}{3} ;+\infty\right[$ par: $f(x)=\dfrac{3 x^{2}-11 x+11}{-3 x+2}$
\begin{enumMet}
	\item Déterminer l'expression de $f^{\prime}$\quad \bareme{2}\\
	\begin{corrBox}
		$f^{\prime}(x)=\dfrac{(6 x-11)(-3 x+2)+3\left(3 x^{2}-11 x+11\right)}{(-3 x+2)^{2}}$
		
		$\phantom{f^{\prime}(x)}=\dfrac{-18 x^{2}-22+33 x+12 x+9 x^{2}-33 x+33}{(-3 x+2)^{2}}$
		
		$\phantom{f^{\prime}(x)}=\dfrac{-9 x^{2}+12 x+11}{(-3 x+2)^{2}}$
	\end{corrBox}
	\item Vérifier que pour tout réel $x$ tel que $x>\dfrac{2}{3}$\\ $f(x)=-x+3+\dfrac{5}{-3 x+2}$\quad \bareme{2}\\
	\begin{corrBox}
		$-x+3+\dfrac{5}{-3 x+2}=\dfrac{(-x+3)(-3 x+2)+5}{-3 x+2}=\dfrac{3 x^{2}-11 x+11}{-3 x+2}=f(x)$
	\end{corrBox}
	\item En déduire une autre expression de $f^{\prime}$\quad \bareme{2}\\
	\begin{corrBox}
		$f^{\prime}(x)=-1-\dfrac{5 \times(-3)}{(-3 x+2)^{2}}=\dfrac{-(-3 x+2)^{2}+15}{(-3 x+2)^{2}}=\dfrac{-9 x^{2}+12 x-4+15}{(-3 x+2)^{2}}=\dfrac{-9 x^{2}+12 x+11}{(-3 x+2)^{2}}$
	\end{corrBox}
	\item Vérifier la cohérence des résultats obtenus au 1) et au 3).\quad \bareme{1}\\
	\begin{corrBox}
		Les résultats des 2 réponses sont identiques.
	\end{corrBox}
\end{enumMet}
