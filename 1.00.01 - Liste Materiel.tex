\section*{Pour celles et ceux qui travaillent avec des cahiers}

\begin{itemMet}
	\item 1 cahier format A4 petits carreaux 192 pages pour le cours 
	\item 1 cahier format A4 petits carreaux 192 pages pour les exercices 
	\item 1 cahier format A4 petits carreaux 96 pages pour les séances informatiques 
	\item 1 pochette pour conserver les interrogations et leur correction Règle graduée, équerre, compas, rapporteur, crayon à papier, gomme, etc... 
	\item 1 paquet de feuilles simples format $A 4$ petits carreaux avec marge pour les interrogations 
	\item 1 paquet de feuilles doubles format A4 petits carreaux avec marge pour les interrogations 
	\item 1 calculatrice graphique Mode Examen avec Python (Numworks)
\end{itemMet}

\section*{Pour celles et ceux qui travaillent avec des classeurs}

\begin{itemMet}
	\item 1 classeur format A4 suffisamment épais
	\item 4 intercalaires pour : cours / exercices / séances informatiques / interrogations et corrections 
	\item Règle graduée, équerre, compas, rapporteur, crayon à papier, gomme, etc...
	\item 1 paquet de feuilles simples format A4 petits carreaux avec marge pour les interrogations
	\item 1 paquet de feuilles doubles format A4 petits carreaux avec marge pour les interrogations 
	\item 1 calculatrice graphique Mode Examen avec Python (Numworks)
\end{itemMet}

\section*{Pour celles et ceux qui travaillent avec des portes-vues}
 
\begin{itemMet}

	\item 1 porte-vues format $A 4$ environ 192 pages pour les exercices
	\item 1 porte-vues format A4 environ 96 pages pour les séances informatiques
	\item 1 pochette pour conserver les interrogations et leur correction

	\item 1 calculatrice graphique Mode Examen avec Python (Numworks)
\end{itemMet}

	\item 1 paquet de feuilles simples format $A 4$ petits carreaux avec marge pour les interrogations 
	\item 1 paquet de feuilles doubles format $A 4$ petits carreaux avec marge pour les interrogations
	\item 1 calculatrice graphique Mode Examen avec Python (Numworks)
\end{itemMet}
