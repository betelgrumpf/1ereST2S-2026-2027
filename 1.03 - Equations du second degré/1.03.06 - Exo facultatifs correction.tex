\exercice{- Racine évidente} 
Factoriser les fonctions polynômes du second degré sans calculer leur discriminant.
\begin{enumMet}
	\item $f: x \mapsto-6 x^2+10 x-4$
	\begin{corrBox}
		$f(1)=-6\times1^2+10\times1-4=-6+10-4=0$. 1 est une racine évidente.\\
		$P=x_1\times x_2=\dfrac{c}{a}=\dfrac{4}{6}=\dfrac{2}{3}\iff x_2=\dfrac{2}{3}$\\
		$f(x)=a(x-x_1)(x-x_2)=-6(x-1)(x-\dfrac{2}{3})$
	\end{corrBox}
	\item $g: x \mapsto 6 x^2+2 x-20$
	\begin{corrBox}
		$f(-2)=6\times(-2)^2+2\times(-2)-20=24-4-20=0$. $-2$ est une racine évidente.\\
		$P=x_1\times x_2=\dfrac{c}{a}=\dfrac{-20}{6}=-\dfrac{10}{3}\iff x_2=-\dfrac{5}{3}$\\
		$f(x)=a(x-x_1)(x-x_2)=6(x+2)(x+\dfrac{5}{3})$
	\end{corrBox}
	\item $h: x \mapsto 4 x^2-14 x+12$
	\begin{corrBox}
		$f(2)=4\times(2)^2-14\times2+12=16-28+12=0$. $2$ est une racine évidente.\\
		$P=x_1\times x_2=\dfrac{c}{a}=\dfrac{12}{4}=3\iff x_2=\dfrac{3}{2}$\\
		$f(x)=a(x-x_1)(x-x_2)=4(x-2)(x-\dfrac{3}{2})$
	\end{corrBox}
	\item $k: x \mapsto 2 x^2-8 x+8$
	\begin{corrBox}
		$f(2)=2\times(2)^2-8\times2+8=8-16+8=0$. $2$ est une racine évidente.\\
		$P=x_1\times x_2=\dfrac{c}{a}=\dfrac{8}{2}=4\iff x_2=2=x_1$\\
		$f(x)=a(x-x_1)(x-x_2)=2(x-2)(x-2)=2(x-2)^2$
	\end{corrBox}
\end{enumMet}

		$\Delta<0$ donc $S=\varnothing$

	\end{corrBox}
	\item $5 x^2-2 x+1=0$

		$\Delta=b^2-4ac=(-2)^2-4\times5\times1=-16$\\
		$\Delta<0$ donc $S=\varnothing$
	\end{corrBox}
	\item $2 x^2+4=-6 x$
	\begin{corrBox}
		$2 x^2+6 x+4=0$\\
		$\Delta=b^2-4ac=6^2-4\times2\times4=4$\\
		$\Delta>0$ donc 2 solutions:\\
		$x_1=\dfrac{-b-\sqrt{\Delta}}{2a}=\dfrac{-6-\sqrt{4}}{2\times2}=-2$\quad et \quad $x_2=\dfrac{-b+\sqrt{\Delta}}{2a}=\dfrac{-6+\sqrt{4}}{2\times2}=-1$\\
		$S=\left\{-2;-1\right\}$
	\end{corrBox}
	\item $x(2 x-1)=1$
	\begin{corrBox}
		$2x^2-x-1=0$\\
		$\Delta=b^2-4ac=(-1)^2-4\times2\times(-1)=9$\\
		$\Delta>0$ donc 2 solutions:\\
		$x_1=\dfrac{-b-\sqrt{\Delta}}{2a}=\dfrac{1-\sqrt{9}}{2\times2}=-\dfrac{1}{2}$\quad et \quad $x_2=\dfrac{-b+\sqrt{\Delta}}{2a}=\dfrac{1+\sqrt{9}}{2\times2}=1$\\
		$S=\left\{-\dfrac{1}{2};1\right\}$
	\end{corrBox}
	\item $x^2=-5 x-1$
	\begin{corrBox}
		$x^2+5x+1=0$\\

		$\Delta>0$ donc 2 solutions:\\

		$S=\left\{\dfrac{-5-\sqrt{21}}{2};\dfrac{-5+\sqrt{21}}{2}\right\}$
	\end{corrBox}
	\item $-x+3 x^2-1=0$
	\begin{corrBox}
		$3x^2-x-1=0$\\
		$\Delta=b^2-4ac=(-1)^2-4\times3\times(-1)=13$\\
		$\Delta>0$ donc 2 solutions:\\
		$x_1=\dfrac{-b-\sqrt{\Delta}}{2a}=\dfrac{1-\sqrt{13}}{6}$\quad et \quad $x_2=\dfrac{-b+\sqrt{\Delta}}{2a}=\dfrac{1+\sqrt{13}}{6}$\\
		$S=\left\{\dfrac{1-\sqrt{13}}{6};\dfrac{1+\sqrt{13}}{6}\right\}$
	\end{corrBox}
	\item $x(8-x)+1=0$
	\begin{corrBox}
		$-x^2+8x+1=0$\\
		$\Delta=b^2-4ac=8^2-4\times(-1)\times1=68$\\
		$\Delta>0$ donc 2 solutions:\\
		$x_1=\dfrac{-b-\sqrt{\Delta}}{2a}=\dfrac{-8-\sqrt{68}}{2\times (-1)}=4+\sqrt{17}$\quad et \quad $x_2=\dfrac{-b+\sqrt{\Delta}}{2a}=\dfrac{-8+\sqrt{68}}{2\times (-1)}=4-\sqrt{17}$\\
		$S=\left\{4-\sqrt{17};4+\sqrt{17}\right\}$
	\end{corrBox}
	\item $2 x^2+6 x+\dfrac{9}{2}=0$
	\begin{corrBox}
		$\Delta=b^2-4ac=6^2-4\times2\times\dfrac{9}{2}=0$\\
		$\Delta=0$ donc 1 solution:\\
		$x_0=\dfrac{-b}{2a}=\dfrac{-6}{4}=-\dfrac{3}{2}$\\
		$S=\left\{-\dfrac{3}{2}\right\}$
	\end{corrBox}
	\item $x^2+2 \sqrt{3} x+3=0$
	\begin{corrBox}
		$\Delta=b^2-4ac=(2 \sqrt{3})^2-4\times1\times3=0$\\
		$\Delta=0$ donc 1 solution:\\
		$x_0=\dfrac{-b}{2a}=\dfrac{-2 \sqrt{3}}{2}=-\sqrt{3}$\\
		$S=\left\{-\sqrt{3}\right\}$
	\end{corrBox}
	\item $-3 x^2+x=-\dfrac{1}{4}$
	\begin{corrBox}
		$-3 x^2+x+\dfrac{1}{4}=0$\\
		$\Delta=b^2-4ac=1^2-4\times(-3)\times\dfrac{1}{4}=4$\\
		$\Delta>0$ donc 2 solutions:\\
		$x_1=\dfrac{-b-\sqrt{\Delta}}{2a}=\dfrac{-1-\sqrt{4}}{2\times (-3)}=\dfrac{1}{2}$\quad et \quad $x_2=\dfrac{-b+\sqrt{\Delta}}{2a}=\dfrac{-1+\sqrt{4}}{2\times (-3)}=-\dfrac{1}{6}$\\
		$S=\left\{-\dfrac{1}{6};\dfrac{1}{2}\right\}$
	\end{corrBox}
	\item $2 x(5+2 x)=9-2 x$

		$S=\left\{\dfrac{-3-3\sqrt{2}}{2};\dfrac{-3+3\sqrt{2}}{2}\right\}$
	\end{corrBox}

\exercice{- Factoriser}
Factoriser $f(x)$ par la méthode la plus adaptée.
%\begin{multicols}{2}
	\begin{enumMet}
		\item $f(x)=137 x^2-x-136$
		\begin{corrBox}
			$f(1)=137\times1^2-1-136=137+-1-136=0$. 1 est une racine évidente.\\
			$P=x_1\times x_2=\dfrac{c}{a}=\dfrac{136}{137}\iff x_2=\dfrac{136}{137}$\\
			$f(x)=a(x-x_1)(x-x_2)=137(x-1)\left(x-\dfrac{136}{137}\right)=(x-1)(137x-136)$
		\end{corrBox}
		\item $f(x)=4 x^2-8 x+4$
		\begin{corrBox}
			$f(1)=4\times1^2-8\times1+4=4-8+4=0$. 1 est une racine évidente.\\
			$P=x_1\times x_2=\dfrac{c}{a}=\dfrac{4}{4}=1\iff x_2=1=x_1$\\
			$f(x)=a(x-x_1)(x-x_2)=4(x-1)^2$
		\end{corrBox}
		\item $f(x)=-x^2+x+12$
		\begin{corrBox}
			$\Delta=49\quad x_1=-3$ et $x_2=4$\\
			$f(x)=a(x-x_1)(x-x_2)=-(x-(-3))(x-4)=-(x+3)(x-4)$
		\end{corrBox}
		\item $f(x)=x^2-7 x+10$
		\begin{corrBox}
			$\Delta=9\quad x_1=2$ et $x_2=5$\\
			$f(x)=a(x-x_1)(x-x_2)=(x-2)(x-5)$
		\end{corrBox}
		\item $f(x)=-x^2+9$
		\begin{corrBox}
			$f(x)=9-x^2=3^2-x^2=(3-x)(3+x)$
		\end{corrBox}
		\item $f(x)=-6 x^2+25 x-14$
		\begin{corrBox}
			$\Delta=289\quad x_1=\dfrac{2}{3}$ et $x_2=\dfrac{7}{2}$\\
			$f(x)=a(x-x_1)(x-x_2)=-6\left(x-\dfrac{2}{3}\right)\left(x-\dfrac{7}{2}\right)=-(3x-2)(2x-7)$
		\end{corrBox}
	\end{enumMet}
