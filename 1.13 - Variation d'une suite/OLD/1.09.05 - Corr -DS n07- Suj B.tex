\exerciceDS{ - Reconnaître une suite arithmétique - ( 4,5 points )}

Reconnaître parmi les suites définies sur $\mathbb{N}$ ci-dessous celles qui sont arithmétiques et préciser alors leur premier terme et leur raison.
\begin{enumMet}
    \item $u_{n}=-1+4 n$\quad\bareme{0,5}\\
    \begin{corrBox}
        On reconnaît une expression d'une suite arithmétique :\\
        $u_n=u_0+n\times r$ avec $u_0=-1$ et $r=4$
    \end{corrBox}
    \item $u_{n}=3 n^{2}+2$\quad\bareme{0,5}\\
    \begin{corrBox}
        $u_0=2\quad\quad\quad u_1=5\quad\quad\quad u_2=14\quad\quad\quad\quad\quad\quad\quad\quad u_2-u_1=9\neq u_1-u_0=3$\\
        $u_n$ n'est donc pas une suite arithmétique.
    \end{corrBox}
    \item $u_{n}=\dfrac{n+4}{3}$\quad\bareme{0,5}\\
    \begin{corrBox}
        On reconnaît une expression d'une suite arithmétique :\\
        $u_n=u_0+n\times r$ avec $u_0=\dfrac{4}{3}$ et $r=\dfrac{1}{3}$
    \end{corrBox}
    \item $u_{n}=\dfrac{1}{n+2}$ \quad\bareme{0,5}\\
    \begin{corrBox}
        $u_0=\dfrac{1}{2}\quad\quad\quad u_1=\dfrac{1}{3}\quad\quad\quad u_2=\dfrac{1}{4}\quad\quad\quad\quad\quad\quad\quad\quad u_2-u_1=-\dfrac{1}{12}\neq u_1-u_0=-\dfrac{1}{6}$\\
        $u_n$ n'est donc pas une suite arithmétique.
    \end{corrBox}
    \item $u_{0}=3$ et $u_{n+1}=u_{n}+5 n$ \quad\bareme{0,5}\\
    \begin{corrBox}
        $u_{n+1}-u_n=u_n+5 n-u_n=5n$. La différence n'est pas constante.\\
        La suite $u_n$ n'est donc pas arithmétique. 
    \end{corrBox}
    \item $u_{0}=4$ et $u_{n+1}=2,5+u_{n}$ \quad\bareme{0,5}\\
    \begin{corrBox}
        $u_{n+1}-u_n=2,5+u_n-u_n=2,5=r$.\\
        La suite $u_n$ est une suite arithmétique de raison $r=2,5$ et de premier terme $u_0=4$. 
    \end{corrBox}
\end{enumMet}

\exerciceDS{ - Calcul des termes d'une suite - ( 3 points )}
Pour chacune des suites définies pour tout entier naturel $n$, calculer les trois termes suivant le premier.
\begin{enumMet}
    \item $\left\{\begin{array}{l}u_{0}=2 \\ u_{n+1}=3u_{n}-5 n\end{array}\right.$\\
    \begin{corrBox}
        $u_1=3\times2-5\times0=1\quad\quad\quad u_2=3\times1-5\times1=-2\quad\quad\quad u_3=3\times(-2)-5\times2=-16$
    \end{corrBox}
    \item $\left\{\begin{array}{l}u_{0}=0 \\ u_{n+1}=2u_{n}^{2}+\dfrac{1}{2 n+1}\end{array}\right.$\\
    \begin{corrBox}
        $u_1=2\times0^2+\dfrac{1}{2\times0+3}=\dfrac{1}{3}\quad\quad\quad u_2=2\times\left(\dfrac{1}{3}\right)^2+\dfrac{1}{2\times1+3}=\dfrac{19}{45}$\\
        $u_3=2\times\left(\dfrac{19}{45}\right)^2+\dfrac{1}{2\times2+3}=\dfrac{7\ 079}{14\ 175}$
    \end{corrBox}
    \item \ding{46} $\left\{\begin{array}{l}u_{0}=3 \\ u_{n}=5 u_{n-1}-3\end{array}\right.$\\
    \begin{corrBox}
        $u_1=5\times3-3=12\quad\quad\quad u_2=5\times12-3=57\quad\quad\quad u_3=5\times57-3=282$
    \end{corrBox}
\end{enumMet}

\exerciceDS{ - monotonie d'une suite - ( 5 points )}
Étudier la monotonie des suites suivantes.
\begin{enumMet}
    \item pour tout entier naturel $n$, $u_{n}=\dfrac{1}{2 n+5}$.\\
    \begin{corrBox}
        $u_{n+1}-u_n=\dfrac{1}{2n+7}-\dfrac{1}{2n+5}=\dfrac{(2n+5)-(2n+7)}{(2n+5)(2n+7)}=\dfrac{-2}{(2n+5)(2n+7)}<0$\\
        La suite $u_n$ est décroissante. 
    \end{corrBox}
    \item Pour tout entier naturel $n, u_{n}=2\times4^{n}$.\\
    \begin{corrBox}
        $u_{n+1}-u_n=2\times4^{n+1}-2\times4^n=2\times4^n\times(4-1)=3\times2\times4^n>0$.\\
        La suite est croissante.
    \end{corrBox}
    \item Pour tout entier naturel $n, v_{n}=0,7^{n}$.\\
    \begin{corrBox}
        $v_{n+1}-v_n=0,7^{n+1}-0,7^n=0,7^n\times(0,7-1)=-0,3\times0,7^n<0$.\\
        La suite est décroissante.
    \end{corrBox}
    \item Pour tout entier naturel $n, w_{n}=(-4)^{n}$.\\
    \begin{corrBox}
        $w_{n+1}-w_n=(-4)^{n+1}-(-4)^n=(-4)^n\times(-4-1)=-5\times(-4)^n$.\\
        Si n est pair, $w_{n+1}-w_n<0$ et Si n est impair, $w_{n+1}-w_n>0$. La suite n'est donc pas monotone.
    \end{corrBox}
    \item \ding{46} Pour tout entier naturel $n, t_{n}=5 n+8$.\\
    \begin{corrBox}
        Pour tout entier $t_{n+1}=5(n+1)+8=5n+13$\\
        Pour tout entier $t_{n+1}-t_{n}=5n+13-(5n+8)=5>0$\\
        La suite $\left(t_{n}\right)$ est croissante.\\
    \end{corrBox}
\end{enumMet}

\exerciceDS{ - terme général d'une suite arithmétique - ( 4 points )}
Les suites suivantes sont arithmétiques de raison $r$. Exprimer $u_{n}$ en fonction de $n$, pour tout $n \in \mathbb{N}$.
\begin{enumMet}
    \item $u_{3}=5$ et $r=4$\\
    \begin{corrBox}
        $u_0=u_3-3r=5-3\times4=-7\Longrightarrow u_n=u_0+nr \Longrightarrow u_n=-7+4n$
    \end{corrBox}
    \item $u_{8}=20$ et $r=-\dfrac{1}{2}$\\
    \begin{corrBox}
        $u_0=u_8-8r=20+8\times\dfrac{1}{2}=24\Longrightarrow u_n=u_0+nr \Longrightarrow u_n=24-\dfrac{1}{2}n$
    \end{corrBox}
    \item \ding{46} $u_{3}=4$ et $r=8$\\
    \begin{corrBox}
        $u_0=u_3-3r=4-3\times8=-20\Longrightarrow u_n=u_0+nr \Longrightarrow u_n=-20+8n$
    \end{corrBox}
    \item $u_{9}=\dfrac{25}{2}$ et $r=-\dfrac{1}{4}$\\
    \begin{corrBox}
        $u_0=u_9-9r=\dfrac{25}{2}+9\times\dfrac{1}{4}=\dfrac{59}{4}\Longrightarrow u_n=u_0+nr \Longrightarrow u_n=\dfrac{59}{4}-\dfrac{1}{4}n$
    \end{corrBox}
\end{enumMet}

\exerciceDS{ - Calcul d'un terme - ( 4 points )}
Pour chacune des suites suivantes, calculer $u_{22}$.
\begin{enumMet}
    \item La suite $\left(u_{n}\right)$ est arithmétique de raison $r=5$ et telle que $u_{7}=15$.\\
    \begin{corrBox}
        $u_{22}=u_{7}+(22-7)\times r=15+(22-7)\times 5=90$
    \end{corrBox}
    \item La suite $\left(u_{n}\right)$ est arithmétique de raison $r=5$ et telle que $u_{28}=34$.\\
    \begin{corrBox}
        $u_{22}=u_{28}+(22-28)\times r=34+(22-28)\times 5=4$
    \end{corrBox}
    \item La suite $\left(u_{n}\right)$ est définie par $\left\{\begin{array}{l}u_{0}=6 \\ u_{n+1}=u_{n}+7\end{array}\right.$\\
    \begin{corrBox}
        $u_{22}=u_{0}+22\times r=6+22\times 7=160$
    \end{corrBox}
    \item \ding{46} La suite $\left(u_{n}\right)$ est définie par $\left\{\begin{array}{l}u_{1}=-4 \\ u_{n+1}=u_{n}-5\end{array}\right.$ pour $n \in \mathbb{N}$.\\
    \begin{corrBox}
        $u_{22}=u_{1}+(22-1)\times r=-4+(22-1)\times -5=-109$
    \end{corrBox}
\end{enumMet}
