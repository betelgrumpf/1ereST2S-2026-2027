\begin{enumMet}
    \item $f: x \mapsto 3x^{2}-6 x+9$
    \item $g: x \mapsto x^{3}-6 x^{2}+9 x-1$
    %\item $f(x)=2x(3x^2-18)$
    \item \ding{46} $h: x \mapsto 4x-5+\dfrac{3}{x-2}$
\end{enumMet}

\exerciceDS{ - Calcul des termes d'une suite - ( 4 points )}
Pour chacune des suites suivantes définies sur $\mathbb{N}$, calculer $u_{0}, u_{1}, u_{2}, u_{3}$ et $u_{10}$.
\begin{enumMet}
    \item \ding{46} $u_{n}=2 n+5$
    \item $u_{n}=\dfrac{n+1}{n+2}+2$
    \item $u_{n}=2n^{2}-5 n+2$
    \item $u_{n}=2^{n}-5$
\end{enumMet}  

\exerciceDS{ - Calcul des termes d'une suite - ( 3 points )}
Pour chacune des suites définies pour tout entier naturel $n$, calculer les trois termes suivant le premier.
\begin{enumMet}
    \item $\left\{\begin{array}{l}u_{0}=2 \\ u_{n+1}=2u_{n}-6 n\end{array}\right.$
    \item $\left\{\begin{array}{l}u_{0}=0 \\ u_{n+1}=u_{n}^{2}+\dfrac{1}{2 n+3}\end{array}\right.$
    \item \ding{46} $\left\{\begin{array}{l}u_{0}=3 \\ u_{n}=4 u_{n-1}-3\end{array}\right.$
\end{enumMet}

\exerciceDS{ - monotonie d'une suite - ( 5 points )}
Étudier la monotonie des suites suivantes.
\begin{enumMet}
    \item pour tout entier naturel $n$, $u_{n}=\dfrac{1}{2 n+3}$.
    \item Pour tout entier naturel $n, u_{n}=2\times3^{n}$.
    \item Pour tout entier naturel $n, v_{n}=0,6^{n}$.
    \item Pour tout entier naturel $n, w_{n}=(-5)^{n}$.
    \item \ding{46} Pour tout entier naturel $n, t_{n}=2 n+3$.
\end{enumMet}

\exerciceDS{ - terme général d'une suite arithmétique - ( 4 points )}
Les suites suivantes sont arithmétiques de raison $r$. Exprimer $u_{n}$ en fonction de $n$, pour tout $n \in \mathbb{N}$.
\begin{enumMet}
    \item $u_{2}=5$ et $r=2$
    \item $u_{6}=8$ et $r=-\dfrac{1}{2}$
    \item \ding{46} $u_{3}=4$ et $r=8$
    \item $u_{8}=\dfrac{15}{2}$ et $r=-\dfrac{1}{4}$
\end{enumMet}

\exerciceDS{ - Calcul d'un terme - ( 4 points )}
Pour chacune des suites suivantes, calculer $u_{25}$.
\begin{enumMet}
    \item La suite $\left(u_{n}\right)$ est arithmétique de raison $r=2$ et telle que $u_{9}=12$.
    \item La suite $\left(u_{n}\right)$ est arithmétique de raison $r=5$ et telle que $u_{30}=34$.
    \item La suite $\left(u_{n}\right)$ est définie par $\left\{\begin{array}{l}u_{0}=8 \\ u_{n+1}=u_{n}+8\end{array}\right.$
    \item \ding{46} La suite $\left(u_{n}\right)$ est définie par $\left\{\begin{array}{l}u_{1}=-5 \\ u_{n+1}=u_{n}-3\end{array}\right.$ pour $n \in \mathbb{N}$.
\end{enumMet}
