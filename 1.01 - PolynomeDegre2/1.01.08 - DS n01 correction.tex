\exerciceDS{ - identifier la forme d'un trinôme  - (3 points)}

$f$ est une fonction polynôme du second degré. Dans chaque cas, préciser la forme de l'expression (développée ou canonique) et préciser les valeurs $a, b, c$, ou $a, \alpha, \beta$.
\begin{multicols}{2}
	\begin{enumMet}
		\item $f(x)=4 x+2 x^2-1$ \quad \bareme{0,5}
		\begin{corrBox}
			Forme développée. $a=2$, $b=4$ et $c=-1$
		\end{corrBox}
		\item $f(x)=3+4x-5x^2$ \quad \bareme{0,5}
		\begin{corrBox}
			Forme développée. $a=-5$, $b=4$ et $c=3$
		\end{corrBox}
		\item $f(x)=-15 x^2+3$ \quad \bareme{0,5}
		\begin{corrBox}
			Forme développée. $a=-15$, $b=0$ et $c=3$\\
			Forme canonique. $a=-15$, $\alpha=0$ et $\beta=3$
		\end{corrBox}
		\item $f(x)=-5 x^2$ \quad \bareme{0,5}
		\begin{corrBox}
			Forme développée. $a=-5$, $b=0$ et $c=0$\\
			Forme canonique. $a=-5$, $\alpha=0$ et $\beta=0$
		\end{corrBox}
		\item $f(x)=4-2(x+1)^2$ \quad \bareme{0,5}
		\begin{corrBox}
			Forme canonique. $a=-2$, $\alpha=-1$ et $\beta=4$
		\end{corrBox}
		\item $f(x)=5(x+3)^2-8$ \quad \bareme{0,5}
		\begin{corrBox}
			Forme canonique. $a=5$, $\alpha=-3$ et $\beta=-8$
		\end{corrBox}
	\end{enumMet}
\end{multicols}

\exerciceDS{ - Passer à la forme canonique  - (4 points)}
Transformer les fonctions suivantes pour les écrire sous forme canonique. La transformation de la première fonction devra se faire au moyen de la factorisation.
%\begin{multicols}{2}
    \begin{enumMet}
        \item $f(x)=x^2+4x+5$ ; \quad \bareme{1}
        \begin{corrBox}
        	$f(x)=x^2+4x+5=(x^2+4x)+5=(x^2+4x+4-4)+5=(x+2)^2-4+5=(x+2)^2+1$
        \end{corrBox}
        \item  $g(x)=x^2-5x-4$ ; \quad \bareme{1}
         \begin{corrBox}
        	$g(x)=x^2-5x-4=(x^2-5x)-4=(x^2-5x+6,25-6,25)-4=(x-2,5)^2-6,25-4$\\
        	D'ou $g(x)=(x-2,5)^2-10,25$
        \end{corrBox}
        \item $h(x)=-3x^2+5x+15$ ; \quad \bareme{1}
         \begin{corrBox}
        	$\alpha=\dfrac{-b}{2a}=\dfrac{-5}{2\times (-3)}=\dfrac{5}{6}$\quad \quad\quad \quad $\beta=f(\alpha)=-3\times \left(\dfrac{5}{6}\right)^2+5\times \dfrac{5}{6}+15=\dfrac{205}{12}$\\
        	D'ou $h(x)=-3\left(x-\dfrac{5}{6}\right)^2+\dfrac{205}{12}$
        \end{corrBox}
        \item $k(x)=-7x+2x^2-4$; \quad \bareme{1}
         \begin{corrBox}
         	$\alpha=\dfrac{-b}{2a}=\dfrac{7}{2\times 2}=\dfrac{7}{4}$\quad \quad\quad \quad $\beta=f(\alpha)=2\times \left(\dfrac{7}{4}\right)^2-7\times \dfrac{7}{4}-4=-\dfrac{81}{8}$\\
         	D'ou $k(x)=2\left(x-\dfrac{7}{4}\right)^2-\dfrac{81}{8}$
         \end{corrBox}
    \end{enumMet}

\exerciceDS {- points et paraboles  - (3 points)}
En sachant que la courbe représentative passe par le point de coordonnées $A(2;0)$ et à l'aide du tableau de variations ci-dessous, retrouver l'expression de la fonction $f(x)$. On donnera la forme développée et la forme canonique.

\begin{tikzpicture}
	\tkzTabInit{$x$ / 1 , $f$ / 1.5}{$-\infty$, $6$, $+\infty$}
	\tkzTabVar{-/ , +/ $10$, -/ }
\end{tikzpicture}

\begin{corrBox}
	Dans le tableau de variation, on voit que $\alpha=6$ et $\beta=10$. Donc $f$ est de la forme $f(x)=a(x-6)^2+10$ avec $a<0$.\\
	De plus si $x=2$ alors $f(x)=0$, donc $0=a(2-6)^2+10$ d'ou $a=\dfrac{-10}{16}=-\dfrac{5}{8}$.\quad \bareme{1}\\
	La forme canonique de f est donc : $f(x)=-\dfrac{5}{8}(x-6)^2+10$\quad \bareme{1}\\
	$f(x)=-\dfrac{5}{8}(x^2-12x+36)+10=-\dfrac{5}{8}x^2+\dfrac{15}{2}x-\dfrac{45}{2}+10=-\dfrac{5}{8}x^2+\dfrac{15}{2}x-\dfrac{25}{2}$\quad \bareme{1}
\end{corrBox}

\exerciceDS {- graphiques et paraboles - (5 points)}
$f$, $g$, $h$ sont définies sur \R par :
\begin{multicols}{3}
	\begin{enumMet}
		\item $f(x)=-0,25x^2-0,5x+2$
		\item $g(x)=4x^2-5x-1$
		\item $h(x)=5(x-3)^2+1$
		\item $m(x)=-5x^2+30x-44$
		\item $p(x)=2(x+3)^2-1$
	\end{enumMet}
\end{multicols}
Associer chaque courbe à sa fonction associée en expliquant la démarche.

\begin{corrBox}
	$f$ et $m$ correspondent à $C_1$ et $C_3$ car $a<0$ et les 2 paraboles sont orientés vers le bas.\\
	Pour $f(x)$, on a $\alpha=-1$ donc la courbe représentative de $f$ est $C_1$.\quad \bareme{1}\\
	Pour $m(x)$, on a $\alpha=3$ donc la courbe représentative de $m$ est $C_3$\quad \bareme{1}\\

	pour $g(x)$, on a $\alpha=\dfrac{5}{8}$ donc la courbe représentative de $g$ est $C_4$.\quad \bareme{1}
\end{corrBox}
