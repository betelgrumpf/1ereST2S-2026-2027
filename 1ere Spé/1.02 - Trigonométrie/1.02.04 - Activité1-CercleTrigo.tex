\begin{spacing}{1.7}
\begin{enumMet}
	
	\item Quelle est la longueur parcourue par la bille lorsqu'elle revient à son point de départ? (Penser au périmètre d'un cercle):\pointille[12cm]
	\item On considère les points $I^{\prime}$ et $J$ 'symétriques respectifs de $I$ et $J$ par rapport à l'origine du repère $\mathrm{O}$.Quelle est la distance parcourue par la bille de son point de départ au point $I^{\prime}$ ? (Autrement dit : quelle est la longueur de l'arc de cercle $\wideparen{II'}$ ?)\pointille[12cm]
	\item Quelle est la longueur de l'arc de cercle $\wideparen{IJ}$? (c'est la distance parcourue par la bille de son point de départ au point J.)\pointille[12cm]
	\item Le point $A$ est situé au tiers de l'arc de cercle $\wideparen{IJ}$. La longueur de l'arc $\wideparen{IA}$ est \pointille[3cm].\\
	Le point $B$ est situé à la moitié de l'arc de cercle $\wideparen{IJ}$. La longueur de l'arc $\wideparen{IB}$ est \pointille[3cm].\\
	Le point $C$ est situé au deux tiers de l'arc de cercle $\wideparen{IJ}$. La longueur de l'arc $\wideparen{IC}$ est \pointille[3cm].
	\begin{bdefi}
	la mesure en radian, notée rad, de l'angle géométrique $\widehat{IOA}$ est égale à la longueur de l'arc de cercle $\wideparen{IA}$. On a donc $\widehat{IOA}=$\pointille[3cm] rad.
	\end{bdefi}
	\begin{brmq}
		\grasInRem{Attention}: cette égalité entre mesure d'angle en radian et longueur d'arc de cercle n'est vraie que pour un cercle de rayon 1 .
	\end{brmq}
