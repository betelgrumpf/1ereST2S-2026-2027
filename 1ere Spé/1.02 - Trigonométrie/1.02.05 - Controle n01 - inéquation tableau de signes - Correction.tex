	Résoudre les inéquations suivantes à l'aide d'un tableau de signes :
	\ \\
	\ \\
	\begin{enumMet}
		\item $$\dfrac{5x-6}{2x-5}\geqslant0\ \ \text{(2 points)}$$
		\begin{corrBox}
			$$5x-6=0 \iff x=\dfrac{6}{5}$$
			
			$$2x-5=0 \iff x=\dfrac{5}{2}$$
			\begin{tikzpicture}
				\tkzTabInit{$x$ / 1 , $5x-6$ / 1, $2x-5$ / 1, $\dfrac{5x-6}{2x-5}$ / 1}{$-\infty$, $\dfrac{6}{5}$, $\dfrac{5}{2}$, $+\infty$}
				\tkzTabLine{, -, z, +, t, +, }
				\tkzTabLine{, -, t, -, z, +, }
				\tkzTabLine{, +, z, -, d, +, }
			\end{tikzpicture}
		$$S=\left]-\infty;\dfrac{6}{5}\right]\cup\left]\dfrac{5}{2};+\infty\right[$$	
		\end{corrBox}
		\item $$(-2x+9)(-3x-7)<0\ \ \text{(2 points)}$$
		\begin{corrBox}
			$$-2x+9=0 \iff x=\dfrac{9}{2}$$
			
			$$-3x-7=0 \iff x=-\dfrac{7}{3}$$
			\begin{tikzpicture}
				\tkzTabInit[lgt = 4]{$x$ / 1 , $-2x+9$ / 1, $-3x-7$ / 1, $(-2x+9)(-3x-7)$ / 1}{$-\infty$, $-\dfrac{7}{3}$, $\dfrac{9}{2}$, $+\infty$}
				\tkzTabLine{, +, t, +, z, -, }
				\tkzTabLine{, +, z, -, t, -, }
				\tkzTabLine{, +, z, -, z, +, }
			\end{tikzpicture}
			$$S=\left]-\dfrac{7}{3};\dfrac{9}{2}\right[$$	
		\end{corrBox}
		\newpage
		\item $$\dfrac{(-3x+4)(-9x+5)}{2x-7}\leqslant0\ \ \text{(3 points)}$$
		\begin{corrBox}
			$$-3x+4=0 \iff x=\dfrac{4}{3}$$
			$$-9x+5=0 \iff x=\dfrac{5}{9}$$
			$$2x-7=0 \iff x=\dfrac{7}{2}$$
			\begin{tikzpicture}
				\tkzTabInit[lgt = 4]{$x$ / 1 , $-3x+4$ / 1, $-9x+5$ / 1, $2x-7$ / 1, $\dfrac{(-3x+4)(-9x+5)}{2x-7}$ / 1}{$-\infty$, $\dfrac{5}{9}$, $\dfrac{4}{3}$,$\dfrac{7}{2}$, $+\infty$}
				\tkzTabLine{, +, t, +, z, -, t, -, }
				\tkzTabLine{, +, z, -, t, -, t, -, }
				\tkzTabLine{, -, t, -, t, -, z, +, }
				\tkzTabLine{, -, z, +, z, -, d, +, }
			\end{tikzpicture}
			$$S=\left]-\infty;\dfrac{5}{9}\right]\cup\left[\dfrac{4}{3};\dfrac{7}{2}\right[$$	
		\end{corrBox}
		\item $$\dfrac{2x-9}{(-4x-8)(3x-7)}\geqslant0\ \ \text{(3 points)}$$
		\begin{corrBox}
			$$2x-9=0 \iff x=\dfrac{9}{2}$$
			$$-4x-8=0 \iff x=-2$$
			$$3x-7=0 \iff x=\dfrac{7}{3}$$
			\begin{tikzpicture}
				\tkzTabInit[lgt = 4]{$x$ / 1 , $2x-9$ / 1, $-4x-8$ / 1, $3x-7$ / 1, $\dfrac{2x-9}{(-4x-8)(3x-7)}$ / 1}{$-\infty$, $-2$, $\dfrac{7}{3}$,$\dfrac{9}{2}$, $+\infty$}
				\tkzTabLine{, -, t, -, t, -, z, +, }
				\tkzTabLine{, +, z, -, t, -, t, -, }
				\tkzTabLine{, -, t, -, z, +, t, +, }
				\tkzTabLine{, +, d, -, d, +, z, -, }
			\end{tikzpicture}
			$$S=\left]-\infty;-2\right[\cup\left]\dfrac{7}{3};\dfrac{9}{2}\right]$$
		\end{corrBox}
	\end{enumMet}

	Résoudre les inéquations suivantes à l'aide d'un tableau de signes :
	\ \\
	\ \\
	\begin{enumMet}
		\item $$\dfrac{3x-6}{2x-3}\leqslant0\ \ \text{(2 points)}$$
		\begin{corrBox}
			$$3x-6=0 \iff x=2$$
			
			$$2x-3=0 \iff x=\dfrac{3}{2}$$
			\begin{tikzpicture}
				\tkzTabInit{$x$ / 1 , $3x-6$ / 1, $2x-3$ / 1, $\dfrac{3x-6}{2x-3}$ / 1}{$-\infty$, $\dfrac{3}{2}$, $2$, $+\infty$}
				\tkzTabLine{, -, t, -, z, +, }
				\tkzTabLine{, -, z, +, t, +, }
				\tkzTabLine{, +, d, -, z, +, }
			\end{tikzpicture}
			$$S=\left]\dfrac{3}{2};2\right]$$	
		\end{corrBox}
		\item $$(-2x+7)(3x-5)>0\ \ \text{(2 points)}$$
		\begin{corrBox}
			$$-2x+7=0 \iff x=\dfrac{7}{2}$$
			
			$$3x-5=0 \iff x=\dfrac{5}{3}$$
			\begin{tikzpicture}
				\tkzTabInit[lgt = 4]{$x$ / 1 , $-2x+7$ / 1, $3x-5$ / 1, $(-2x+7)(3x-5)$ / 1}{$-\infty$, $\dfrac{5}{3}$, $\dfrac{7}{2}$, $+\infty$}
				\tkzTabLine{, +, t, +, z, -, }
				\tkzTabLine{, -, z, +, t, +, }
				\tkzTabLine{, -, z, +, z, -, }
			\end{tikzpicture}
			$$S=\left]\dfrac{5}{3};\dfrac{7}{2}\right[$$	
		\end{corrBox}
	\newpage
		\item $$\dfrac{(-3x+2)(-5x-5)}{2x-5}\geqslant0\ \ \text{(3 points)}$$
		\begin{corrBox}
			$$-3x+2=0 \iff x=\dfrac{2}{3}$$
			$$-5x-5=0 \iff x=-1$$
			$$2x-5=0 \iff x=\dfrac{5}{2}$$
			\begin{tikzpicture}
				\tkzTabInit[lgt = 4]{$x$ / 1 , $-3x+2$ / 1, $-5x-5$ / 1, $2x-5$ / 1, $\dfrac{(-3x+2)(-5x-5)}{2x-5}$ / 1}{$-\infty$, $-1$, $\dfrac{2}{3}$,$\dfrac{5}{2}$, $+\infty$}
				\tkzTabLine{, +, t, +, z, -, t, -, }
				\tkzTabLine{, +, z, -, t, -, t, -, }
				\tkzTabLine{, -, t, -, t, -, z, +, }
				\tkzTabLine{, -, z, +, z, -, d, +, }
			\end{tikzpicture}
			$$S=\left[-1;\dfrac{2}{3}\right]\cup\left]\dfrac{5}{2};+\infty\right[$$	
		\end{corrBox}
		\item $$\dfrac{2x-6}{(-3x-6)(5x-7)}\leqslant0\ \ \text{(3 points)}$$
		\begin{corrBox}
			$$2x-6=0 \iff x=3$$
			$$-3x-6=0 \iff x=-2$$
			$$5x-7=0 \iff x=\dfrac{7}{5}$$
			\begin{tikzpicture}
				\tkzTabInit[lgt = 4]{$x$ / 1 , $2x-6$ / 1, $-3x-6$ / 1, $5x-7$ / 1, $\dfrac{2x-6}{(-3x-6)(5x-7)}$ / 1}{$-\infty$, $-2$, $\dfrac{7}{5}$,$3$, $+\infty$}
				\tkzTabLine{, -, t, -, t, -, z, +, }
				\tkzTabLine{, +, z, -, t, -, t, -, }
				\tkzTabLine{, -, t, -, z, +, t, +, }
				\tkzTabLine{, +, d, -, d, +, z, -, }
			\end{tikzpicture}
			$$S=\left]-2;\dfrac{7}{5}\right[\cup\left[3;+\infty\right[$$	
		\end{corrBox}
	\end{enumMet}
