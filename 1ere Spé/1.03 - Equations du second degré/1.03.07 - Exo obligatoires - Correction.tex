\section*{Equations du second degré}

\exercice{}

Pour chaque fonction trinôme $f$, dire si la valeur $a$ proposée est une racine de $f$.
\begin{enumMet}
	\item $f(x)=x^2-x+1$ et $a=1$
	\begin{corrBox}
		$f(a)=f(1)=1^2-1+1=1\neq0$
		
		1 n'est pas une racine de f		
	\end{corrBox}
	\item $f(x)=3 x^2+x-2$ et $a=-1$
	\begin{corrBox}
		$f(a)=f(-1)=3\times (-1)^2+(-1)-2=0$
		
		-1 est une racine de f.		
	\end{corrBox}
	\item $f(x)=2 x^2-3 x-2$ et $a=2$
	\begin{corrBox}
		$f(a)=f(2)=2\times 2^2-3\times 2-2=0$
		
		2 est une racine de f.		
	\end{corrBox}
	\item $f(x)=-2 x^2+x+1$ et $a=3$
	\begin{corrBox}
		$f(a)=f(3)=-2\times 3^2+3+1=-14\neq0$
		
		3 n'est pas une racine de f		
	\end{corrBox}
\end{enumMet}

\exercice{}

Calculer le discriminant de chaque trinôme et donner le nombre de racines du trinôme.
\begin{enumMet}
	\item $f(x)=x^2+x-1$
	\begin{corrBox}
		$a=1$ $b=1$ et $c=-1$
		
		$\Delta=b^2-4ac=1^2-4\times 1\times (-1)=5>0$
		
		Le discriminant est $>0$, il y a 2 racines.		
	\end{corrBox}
	\item $g(x)=2 x^2-x+3$
	\begin{corrBox}
		$a=2$ $b=-1$ et $c=3$
		
		$\Delta=b^2-4ac=(-1)^2-4\times 2\times 3=-23<0$
		
		Le discriminant est $<0$, il n'y a pas de racine.		
	\end{corrBox}
	\item $h(x)=-5 x^2+4 x+3$
	\begin{corrBox}
		$a=-5$ $b=4$ et $c=3$
		
		$\Delta=b^2-4ac=4^2-4\times (-5)\times 3=76>0$
		
		Le discriminant est $>0$, il y a 2 racines		
	\end{corrBox}
	\item $j(x)=x^2-\sqrt{2} x+7$
	\begin{corrBox}
		$a=1$ $b=-\sqrt{2}$ et $c=7$
		
		$\Delta=b^2-4ac=(-\sqrt{2})^2-4\times 1\times 7=-26<0$
		
		Le discriminant est $<0$, il n'y a pas de racine.
	\end{corrBox}
\end{enumMet}

\exercice{ - En Physique}
Lors d'un freinage d'urgence, la distance que parcourt le véhicule avant l'arrêt total se décompose en deux parties : la distance parcourue pendant le temps de réaction du conducteur et la distance de freinage parcourue au cours du freinage du véhicule.
\begin{enumMet}
	\item Le temps de réaction du conducteur, c'est-à-dire le temps nécessaire pour prendre conscience de la situation et appuyer sur le frein, est d'environ une seconde. Si on appelle $v$ la vitesse du véhicule en $\mathrm{km} \cdot \mathrm{h}^{-1}$, montrer que la distance $d_r$, en mètre, parcourue pendant ce temps de réaction vérifie : $d_r=\dfrac{v}{3,6}$.
	\begin{corrBox}
		On utilise la formule $d=\dfrac{v}{t}$ en mettant la vitesse en $m.s^{-1}$ (on multiplie par 1000 et on divise par 3600 pour passer de $km.h^{-1}$ à $m.s^{-1}$).
		On remplace t par 1 (le temps de réaction). On retrouve alors : $d_r=\dfrac{v}{3,6}$
	\end{corrBox}
	\item Pour la distance de freinage $d_f$, exprimée en mètre, on donne la formule suivante $d_f=\dfrac{v^2}{200}$.
	La distance d'arrêt est donc égale à $d_a=d_r+d_f$.
	Une voiture roule à $110 \mathrm{~km} \cdot \mathrm{~h}^{-1}$. Quelle est sa distance d'arrêt (on arrondira au centième près)?
	\begin{corrBox}
		$d_a=d_r+d_f=\dfrac{v}{3,6}+\dfrac{v^2}{200}=\dfrac{110}{3,6}+\dfrac{110^2}{200}\approx91,06$m.		
	\end{corrBox}
	\item Quelles sont les vitesses qui permettent de s'arrêter en moins de 15 m ?
	\begin{corrBox}
		On résous l'inéquation $d_a<15m$ soit $\dfrac{v^2}{200}+\dfrac{v}{3,6}-15<0$
		
		$\Delta=\dfrac{611}{1620}$
		
		$v_1\approx-89,19$ et $v_2\approx33,64$
		
		la fonction est négative entre les 2 racines (et $v<0$ n'a pas de sens ), il faut donc une vitesse comprise entre 0 et 33,64$km.h^{-1}$.	
	\end{corrBox}
\end{enumMet}
