\exerciceDS{ - Factorisation - ( 5,5 points )}

Donner les racines du polynôme du second degré donné, puis le factoriser

%    \item \ding{46} $B(x)=-x^{2}+2 x+15$
%    \item $C(x)=x^{2}+18 x+77$
%    \item $D(x)=2 x^{2}-7 x+8$
%\end{enumMet}

\begin{multicols}{2}
	\begin{enumMet}
		\item $A(x)=x^{2}+6 x+9$ \quad \bareme{1}\\
		\begin{corrBox}
			$A(x)=x^{2}+6 x+9=(x+3)^2$\\
			(Identité remarquable)
		\end{corrBox}
		\item \ding{46} $B(x)=-x^{2}+2 x+15$ \quad \bareme{1,5}\\
		\begin{corrBox}
			$B(x)=-x^{2}+2 x+15$\\
			$\Delta=4+4\times15=64$\\
			$x_1=\dfrac{-2-\sqrt{64}}{2\times(-1)}=5$\\
			et $x_2=\dfrac{-2+\sqrt{64}}{2\times(-1)}=-3$\\
			$B(x)=-(x+3)(x-5)$
		\end{corrBox}
		
		\columnbreak
		\item $C(x)=x^{2}+18 x+77$ \quad \bareme{1,5}\\
		\begin{corrBox}
			$C(x)=x^{2}+18 x+77$\\
			$\Delta=18^2-4\times77=16$\\
			$x_1=\dfrac{-18-\sqrt{16}}{2}=-11$\\
			et $x_2=\dfrac{-18+\sqrt{16}}{2}=-7$\\
			$C(x)=(x+11)(x+7)$
		\end{corrBox}
		\item $D(x)=2 x^{2}-7 x+8$\quad \bareme{1,5}\\
		\begin{corrBox}
			$D(x)=2 x^{2}-7 x+8$\\
			$\Delta=(-7)^2-4\times2\times8=-15$\\
			Pas de racines et factorisation impossible.
		\end{corrBox}
	\end{enumMet}
\end{multicols}

\exerciceDS{ - Problème - ( 4 points )}

Une athlète lance un javelot à l'instant $t=0$. La hauteur $h(t)$, en mètre, à l'instant $t$, en seconde, du centre de gravité est :
$$
h(t)=-\frac{1}{2} t^2+8 t+2
$$

La hauteur est mesurée à partir du sol.
\begin{enumMet}
	\item À quel instant le javelot est-il au plus haut et quelle est la hauteur maximale? \quad \bareme{1}
	\begin{corrBox}
		On cherche $\alpha$ et $\beta$:\\
		$\alpha=-\dfrac{b}{2a}=\dfrac{8}{2\times (-0,5)}=8$\quad\quad et \quad\quad 	$\beta=f(\alpha)=-\dfrac{1}{2}\times 8^2+8\times 8+2=34$\\
		Au bout de 8 secondes, le javelot atteint une hauteur de 34m.
	\end{corrBox}
	\item Pendant combien de temps le javelot est-il au-dessus de 32m? \quad \bareme{2}
	\begin{corrBox}
		on résout l'inéquation:
		$h(t)\geqslant32 \iff -\dfrac{1}{2} t^2+8 t+2\geqslant32 \iff -\dfrac{1}{2} t^2+8 t-30\geqslant0$\\
		$\Delta=64-4\times\dfrac{1}{2}\times30 =4$\\
		$t_1=\dfrac{-8-\sqrt{4}}{2\times(-0,5)}=10$\\
		et $t_2=\dfrac{-8+\sqrt{4}}{2\times(-0,5)}=6$\\
		Le javelot sera au-dessus de 32m pendant $x_1-x_2=4$ secondes.
	\end{corrBox}
	\item Au bout de combien de temps le javelot touchera-t-il le sol? (À quel instant le javelot touchera-t-il le sol?) \quad \bareme{1}
	\begin{corrBox}
		on résout l'équation $h(x)=0$:\\
		$-\dfrac{1}{2} t^2+8 t+2=0$
		$\Delta=64-4\times\dfrac{-1}{2}\times2 =68$\\
		$t_1=\dfrac{-8-\sqrt{68}}{2\times(-0,5)}=8-2\sqrt{17}\approx-0,246$\\
		et $t_2=\dfrac{-8+\sqrt{68}}{2\times(-0,5)}=8+2\sqrt{17}\approx 16,246$\\
		Le javelot touchera donc le sol au bout de $16,2$ secondes environ.	
	\end{corrBox}
\end{enumMet}
