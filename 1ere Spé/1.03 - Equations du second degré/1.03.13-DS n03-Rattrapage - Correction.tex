\exerciceDS{ $-$ Factorisation $-$ ( 3 points )}

Donner les racines du polynôme du second degré donné, puis le factoriser si cela est possible
%\begin{multicols}{2}
\begin{enumMet}
    \item $A(x)=-3x^{2}-10 x-3$
    \begin{corrBox}
   		$\Delta=b^2-4ac=(-10)^2-4\times 3\times 3=100-36=64=8^2$ 
   		
   		$x_1=\dfrac{-b-\sqrt{\Delta}}{2a}=\dfrac{10-8}{2\times (-3)}=-\dfrac{1}{3}$
   		
   		$x_2=\dfrac{-b+\sqrt{\Delta}}{2a}=\dfrac{10+8}{2\times (-3)}=-3$
   		
   		$A(x)=-3\left(x+\dfrac{1}{3}\right)(x+3)$	
    \end{corrBox}
    \item $B(x)=3x^{2}+4x-4$
    \begin{corrBox}
    	$\Delta=b^2-4ac=(4)^2-4\times 3\times (-4)=16+48=64=8^2$ 
    	
    	$x_1=\dfrac{-b-\sqrt{\Delta}}{2a}=\dfrac{-4-8}{2\times 3}=-2$
    	
    	$x_2=\dfrac{-b+\sqrt{\Delta}}{2a}=\dfrac{-4+8}{2\times 3}=\dfrac{2}{3}$
    	
    	$A(x)=3\left(x-\dfrac{2}{3}\right)(x+2)$	
    \end{corrBox}
\end{enumMet}
