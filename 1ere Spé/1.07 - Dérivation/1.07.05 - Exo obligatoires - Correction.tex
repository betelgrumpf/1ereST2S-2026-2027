\section*{Nombre dérivé}

\exercice{} 

Soit $h$ un nombre réel et soit $f$ la fonction définie sur $\mathbb{R}$ par $f(x)=x^2+1$.
\begin{enumMet}
	\item Calculer $f(1)$ et $f(1+h)$.
	\item Montrer que $f$ est dérivable en 1 et préciser $f^{\prime}(1)$.
	\item À l'aide de la fonction dédiée de la calculatrice, contrôler le résultat du calcul précédent.
\end{enumMet}

\exercice{} 

Soit $f$ la fonction définie sur $\mathbb{R}$ par $f(x)=x^2-x+5$ et $h$ un nombre réel.
\begin{enumMet}
	\item Calculer $f(-2)$ et $f(-2+h)$, puis montrer que $f$ est dérivable en -2 et préciser $f^{\prime}(-2)$.
	\item À l'aide de la fonction dédiée de la calculatrice, contrôler le résultat du calcul précédent.
\end{enumMet}

\exercice{}

Dans chaque cas, montrer que la fonction $f$ est dérivable en $a$ et calculer $f^{\prime}(a)$ :
\begin{enumMet}
	\item $f: x \mapsto \dfrac{1}{x-1}$ et $a=-1$;
	\begin{corrBox}
		$f(-1)=\dfrac{1}{-1-1}=-\dfrac{1}{2}$
		
		$f(-1+h)=\dfrac{1}{-1+h-1}=\dfrac{1}{-2+h}$
		
		$f(-1+h)-f(-1)=\dfrac{1}{-2+h}+\dfrac{1}{2}=\dfrac{2+(-2+h)}{2(-2+h)}=\dfrac{h}{2(-2+h)}$
		
		$\dfrac{f(-1+h)-f(-1)}{h}=\dfrac{h}{2(-2+h)}\times \dfrac{1}{h}=\dfrac{1}{2(-2+h)}$
		
		$\lim \limits_{h \rightarrow 0} \dfrac{f(-1+h)-f(-1)}{h}=\lim \limits_{h \rightarrow 0} \dfrac{1}{2(-2+h)}=-\dfrac{1}{4}$
		
		$-\dfrac{1}{4}$ est un nombre réel donc f est dérivable en $-1$ et $f'(-1)=-\dfrac{1}{4}$		
	\end{corrBox}
	\item $f: x \mapsto \dfrac{2}{x^2}$ et $a=1$;
	\begin{corrBox}
		$f(1)=\dfrac{2}{1^2}=2$
		
		$f(1+h)=\dfrac{2}{(1+h)^2}=\dfrac{2}{1+2h+h^2}$
		
		$f(1+h)-f(1)=\dfrac{2}{1+2h+h^2}-2=\dfrac{2-2(1+2h+h^2)}{1+2h+h^2}=\dfrac{-2h^2-4h}{1+2h+h^2}$
		
		$\dfrac{f(1+h)-f(1)}{h}=\dfrac{-2h^2-4h}{1+2h+h^2}\times \dfrac{1}{h}=\dfrac{-2h-4}{1+2h+h^2}$
		
		$\lim \limits_{h \rightarrow 0} \dfrac{f(1+h)-f(1)}{h}=\lim \limits_{h \rightarrow 0} \dfrac{-2h-4}{1+2h+h^2}=-4$
		
		$-4$ est un nombre réel donc f est dérivable en $-1$ et $f'(1)=-4$		
	\end{corrBox}
	\newpage
	\item $f: x \mapsto x^3$ et $a=2$.
	\begin{corrBox}
		$f(2)=2^3=8$
		\ \\
		
		$f(2+h)=(2+h)^3=(4+4h+h^2)(2+h)=8+8h+2h^2+4h+4h^2+h^3$
		
		$f(2+h)=h^3+6h^2+12h+8$
		\ \\
		
		$f(2+h)-f(2)=h^3+6h^2+12h+8-8 = h^3+6h^2+12h$
		\ \\
		
		$\dfrac{f(2+h)-f(2)}{h}=\dfrac{h^3+6h^2+12h}{h}=h^2+6h+12$
		\ \\
		
		$\lim \limits_{h \rightarrow 0} \dfrac{f(2+h)-f(2)}{h}=\lim \limits_{h \rightarrow 0} h^2+6h+12=12$
		\ \\
		
		$12$ est un nombre réel donc f est dérivable en $2$ et $f'(2)=12$		
	\end{corrBox}
\end{enumMet}

\section*{Equation réduite d'une tangente}

\exercice{}

Soit une fonction $f$ définie et dérivable sur \R telle que $f(2)=3$ et $f^{\prime}(2)=-2$.
Soit $C_f$ sa courbe représentative. Déterminer l'équation réduite de la tangente à $C_f$ au point d'abscisse 2 .

\begin{corrBox}
	on utilise la formule $y=f'(a)(x-a)+f(a)$ avec $a =2$, $f'(a)=f'(2)=-2$ et $f(a)=f(2)=3$
	
	$y=-2(x-2)+3=-2x+4+3=-2x+7$	
\end{corrBox}

\exercice{}

Soit $f$ la fonction définie sur $\mathbb{R}$ par $f(x)=x^2-2 x+1$ de courbe représentative $C_f$.
\begin{enumMet}
	\item Vérifier par le calcul que $f^{\prime}(-1)=-4$ et $f^{\prime}(2)=2$.
	\begin{corrBox}
		\underline{\mRouge{$f^{\prime}(-1)=-4$}}
		\ \\
		
		$f(-1)=(-1)^2-2\times (-1)+1=4$
		\ \\
		
		$f(-1+h)=(-1+h)^2-2(-1+h)+1=h^2-2h+1+2-2h+1=h^2-4h+4$
		\ \\
		
		$f(-1+h)-f(-1)=h^2-4h+4-4 = h^2-4h$
		\ \\
		
		$\dfrac{f(-1+h)-f(-1)}{h}=\dfrac{h^2-4h}{h}=h-4$
		\ \\
		
		$\lim \limits_{h \rightarrow 0} \dfrac{f(-1+h)-f(-1)}{h}=\lim \limits_{h \rightarrow 0} h-4=-4$
		\ \\
		
		$-4$ est un nombre réel donc f est dérivable en $-1$ et $f'(-1)=-4$
		\ \\
		\ \\
		\newpage
		\underline{\mRouge{$f^{\prime}(2)=2$}}
		\ \\
		
		$f(2)=(2)^2-2\times 2+1=1$
		\ \\
		
		$f(2+h)=(2+h)^2-2(2+h)+1=h^2+4h+4-4-2h+1=h^2+2h+1$
		\ \\
		
		$f(2+h)-f(2)=h^2+2h+1-1 = h^2+2h$
		\ \\
		
		$\dfrac{f(2+h)-f(2)}{h}=\dfrac{h^2+2h}{h}=h+2$
		\ \\
		
		$\lim \limits_{h \rightarrow 0} \dfrac{f(2+h)-f(2)}{h}=\lim \limits_{h \rightarrow 0} h+2=2$
		\ \\
		
		$2$ est un nombre réel donc f est dérivable en $2$ et $f'(2)=2$		
	\end{corrBox}
	\item Déterminer l'équation réduite des tangentes à $C_f$ aux points d'abscisses -1 et 2 .
	\begin{corrBox}
		\grasInProp{Tangente en $-1$}
		
		\ \\
		
		on utilise la formule $y=f'(a)(x-a)+f(a)$ avec $a =-1$, $f'(a)=f'(-1)=-4$ et $f(a)=f(-1)=4$
		
		$y=-4(x+1)+4=-4x-4+4=-4x$
		
		\ \\
		\ \\
		\grasInProp{Tangente en 2}
		
		\ \\
		
		on utilise la formule $y=f'(a)(x-a)+f(a)$ avec $a =2$, $f'(a)=f'(2)=2$ et $f(a)=f(2)=1$
		
		$y=2(x-2)+1=2x-4+1=2x-3$			
	\end{corrBox}
	\item Construire $C_f$, puis $\left(T_{-1}\right)$ et $\left(T_2\right)$.
\end{enumMet}

\exercice{}

Soit $f$ une fonction dérivable sur $\mathbb{R}$ et soit $C_f$ sa courbe représentative dans un repère du plan. Le point $A(4 ;-2)$ appartient à la courbe $C_f$.
Sachant que la tangente $T$ à la courbe $C_f$ au point $A$ passe par le point $J(0 ; 1)$, déterminer $f^{\prime}(4)$, puis l'équation de $T$.

\begin{corrBox}
	$f^{\prime}(4)=\dfrac{y_J-y_A}{x_J-x_A}=\dfrac{1-(-2)}{0-4}=-\dfrac{3}{4}$
	
	on utilise la formule $y=f'(a)(x-a)+f(a)$ avec $a =4$, $f'(a)=f'(4)=-\dfrac{3}{4}$ et $f(a)=f(4)=-2$
	
	$y=-\dfrac{3}{4}(x-4)-2=-\dfrac{3}{4}x+3-2=-\dfrac{3}{4}x+1$
\end{corrBox}
