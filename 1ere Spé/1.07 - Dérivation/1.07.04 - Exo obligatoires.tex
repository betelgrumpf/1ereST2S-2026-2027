\section*{Nombre dérivé}

\exercice{} 

Soit $h$ un nombre réel et soit $f$ la fonction définie sur $\mathbb{R}$ par $f(x)=x^2+1$.
\begin{enumMet}
	\item Calculer $f(1)$ et $f(1+h)$.
	\item Montrer que $f$ est dérivable en 1 et préciser $f^{\prime}(1)$.
	\item À l'aide de la fonction dédiée de la calculatrice, contrôler le résultat du calcul précédent.
\end{enumMet}

\exercice{} 

Soit $f$ la fonction définie sur $\mathbb{R}$ par $f(x)=x^2-x+5$ et $h$ un nombre réel.
\begin{enumMet}
	\item Calculer $f(-2)$ et $f(-2+h)$, puis montrer que $f$ est dérivable en -2 et préciser $f^{\prime}(-2)$.
	\item À l'aide de la fonction dédiée de la calculatrice, contrôler le résultat du calcul précédent.
\end{enumMet}

\exercice{}

Dans chaque cas, montrer que la fonction $f$ est dérivable en $a$ et calculer $f^{\prime}(a)$ :
\begin{enumMet}
	\item $f: x \mapsto \dfrac{1}{x-1}$ et $a=-1$;
	\item $f: x \mapsto \dfrac{2}{x^2}$ et $a=1$;
	\item $f: x \mapsto x^3$ et $a=2$.
\end{enumMet}

\section*{Equation réduite d'une tangente}

\exercice{}

Soit une fonction $f$ définie et dérivable sur \R telle que $f(2)=3$ et $f^{\prime}(2)=-2$.
Soit $C_f$ sa courbe représentative. Déterminer l'équation réduite de la tangente à $C_f$ au point d'abscisse 2 .

\exercice{}

Soit $f$ la fonction définie sur $\mathbb{R}$ par $f(x)=x^2-2 x+1$ de courbe représentative $C_f$.
\begin{enumMet}
	\item Vérifier par le calcul que $f^{\prime}(-1)=-4$ et $f^{\prime}(2)=2$.
	\item Déterminer l'équation réduite des tangentes à $C_f$ aux points d'abscisses -1 et 2 .
	\item Construire $C_f$, puis $\left(T_{-1}\right)$ et $\left(T_2\right)$.
\end{enumMet}

\exercice{}

Soit $f$ une fonction dérivable sur $\mathbb{R}$ et soit $C_f$ sa courbe représentative dans un repère du plan. Le point $A(4 ;-2)$ appartient à la courbe $C_f$.
Sachant que la tangente $T$ à la courbe $C_f$ au point $A$ passe par le point $J(0 ; 1)$, déterminer $f^{\prime}(4)$, puis l'équation de $T$.

\section*{Fonctions dérivées}

\exercice{}

Pour chacune des fonctions suivantes définies sur l'intervalle I, déterminer sa dérivée.
\begin{enumMet}
	\item $f(x)=3 x^2-4 x+3 ; \mathrm{I}=\mathbb{R}$
	\item $f(x)=-4 x^4+3 x^3-2 x^2+x ; \mathrm{I}=\mathbb{R}$
	\item $f(x)=x^3-2 x^2+3 x-4 ; \mathrm{I}=\mathbb{R}$
	\item $f(x)=\sqrt{x}+x ; \mathrm{I}=[0 ;+\infty[$
	\item $f(x)=x^2-\dfrac{1}{x} ; \mathrm{I}=\mathbb{R}^*$		
\end{enumMet}

\exercice{}

Pour chacune des fonctions suivantes définies sur l'intervalle I, déterminer sa dérivée.
\begin{enumMet}
	\item $f(x)=\dfrac{1}{x}\left(x^3-1\right) ; \mathrm{I}=\mathbb{R}^*$
	\item $f(x)=x^2(\sqrt{x}+1) ; \mathrm{I}=[0 ;+\infty[$	
\end{enumMet}

\exercice{}

Pour chacune des fonctions suivantes définies sur l'intervalle I, déterminer sa dérivée.
\begin{enumMet}
	\item $f(x)=\dfrac{1}{x^2+1} ; \mathrm{I}=\mathbb{R}$
	\item $\left.f(x)=\dfrac{1}{\sqrt{x}} ; \mathrm{I}=\right] 0 ;+\infty[$
	
\end{enumMet}

\exercice{}

Pour chacune des fonctions suivantes définies sur l'intervalle I, déterminer sa dérivée.
\begin{enumMet}
	\item $f(x)=\dfrac{x+1}{x-2} ; \mathrm{I}=\mathbb{R} \backslash\{2\}$
	\item $f(x)=\dfrac{x^3+1}{x^2-1} ; \mathrm{I}=\mathbb{R} \backslash\{-1 ; 1\}$
	\item $\left.f(x)=\dfrac{x^2+x+1}{\sqrt{x}} ; \mathrm{I}=\right] 0 ;+\infty[$
\end{enumMet}

\exercice{}

Pour chacune des fonctions suivantes définies sur l'intervalle I, déterminer sa dérivée.
\begin{enumMet}
	\item $f(x)=(1-2 x)^4 ; \mathrm{I}=\mathbb{R}$
	\item $f(x)=\sqrt{2 x+1} ; \mathrm{I}=\left[\frac{-1}{2} ;+\infty[\right.$
\end{enumMet}
