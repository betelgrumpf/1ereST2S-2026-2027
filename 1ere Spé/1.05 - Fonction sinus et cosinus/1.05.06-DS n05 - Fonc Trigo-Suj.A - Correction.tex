	\exerciceDS{ - ( 3 points )}
	Soit $f$ la fonction définie sur \R par::\\
	$f(x)=\sin^2(x)+\cos(x)$.
	\begin{enumMet}
		\item Montrer que la fonction est $2\pi$ périodique.
		\begin{corrBox}
			$f(x+2\pi)=\sin^2(x+2\pi)+\cos(x+2\pi)=\sin^2(x)+\cos(x)=f(x)$.
		\end{corrBox}
		\item Etudier la parité de la fonction f
		\begin{corrBox}
			$f(-x)=\sin^2(-x)+\cos (-x)=(-\sin(x))^2+\cos(x)=\sin^2(x)+\cos(x)$.\\
			On a bien $f(-x)=f(x)$ donc la fonction $f$ est paire.	
		\end{corrBox}
		\item Que peut-on en déduire quant à la représentation graphique de la fonction f?
		\begin{corrBox}
			La courbe représentative de $f$ a une symétrie d'axe l'axe des ordonnées.	
		\end{corrBox}
	\end{enumMet}

	Soit $g$ la fonction définie sur \R par:\\
	$g(x)=2\sin(x)\cos(x)$.
	\begin{enumMet}
		\item \ding{46} Montrer que la fonction est $\pi$ périodique.
		\begin{corrBox}
			$g(x+\pi)=2\sin(x+\pi)\cos(x+\pi)=2\times (-\sin(x))\times (-\cos(x))=2\sin(x)\cos(x)=g(x)$.
		\end{corrBox}
		\item \ding{46} Etudier la parité de la fonction g
		\begin{corrBox}
			$g(-x)=2\sin(-x)\cos(-x)=2\times (-\sin(x))\times \cos(x)=-2\sin(x)\cos(x)=-g(x)$\\
			On a $g(-x)=-g(x)$, la fonction $g$ est donc impaire.
		\end{corrBox}
		\item \ding{46} Que peut-on en déduire quant à la représentation graphique de la fonction g?
		\begin{corrBox}
			La courbe représentative de $g$ a une symétrie centrale avec l'origine du repère comme centre de symétrie.	
		\end{corrBox}
	\end{enumMet}

\begin{multicols}{2}
	\begin{itemMet}
		\item $f(x)=2\sin(2x+\pi)$
		\item $g(x)=2\sin(2x-\dfrac{\pi}{2})$
		\item $h(x)=2\cos(2x-\dfrac{\pi}{2})$
		\item $k(x)=2\cos(2x)$
	\end{itemMet}
\end{multicols}

\begin{corrBox}
	\begin{itemMet}
		\item $f(0)=2\sin(\pi)=0$ et $f(\dfrac{\pi}{4})=2\sin(\dfrac{3\pi}{2})=-2$. Donc la courbe représentative de f passe par le point de coordonnées $(0;0)$ et $\left(\dfrac{\pi}{4};-2\right)$. Il s'agit de $C_4$
		\item $g(0)=2\sin(\dfrac{-\pi}{2})=-2$ et $g(\dfrac{\pi}{4})=2\sin(0)=0$. Donc la courbe représentative de g passe par le point de coordonnées $(0;-2)$ et $\left(\dfrac{\pi}{4};0\right)$. Il s'agit de $C_2$.
		\item $h(0)=2\cos(\dfrac{-\pi}{2})=0$ et $h(\dfrac{\pi}{4})=2\cos(0)=2$. Donc la courbe représentative de h passe par le point de coordonnées $(0;0)$ et $\left(\dfrac{\pi}{4};2\right)$. Il s'agit de $C_1$
		\item $k(0)=2\cos(0)=2$ et $k(\dfrac{\pi}{4})=2\cos(\dfrac{\pi}{2})=0$. Donc la courbe représentative de k passe par le point de coordonnées $(0;2)$ et $\left(\dfrac{\pi}{4};0\right)$. Il s'agit de $C_3$
	\end{itemMet}
\end{corrBox}

\exerciceDS{ - (6 points)}

Résoudre les équations ci-dessous dans l'intervalle I donné.
\begin{enumMet}
	\item $\sin (x)=\dfrac{1}{2}$ et $I=\left[-\pi ; \pi\right]$.
	\begin{corrBox}
		On commence par étudier la solution dans l'intervalle\\ $]-\pi;\pi]$ (parmi les mesures principales de l'angle).\\
		L'équation $\sin (x)=\dfrac{1}{2}$ a pour solution :\\
		$\begin{cases}
			x_1 &=\dfrac{\pi}{6}\\
			x_2 &=\dfrac{5\pi}{6}
		\end{cases}$\\
		$x_1$ et $x_2$ sont les solutions dans l'intervalle $\left[-\pi ; \pi\right]$\\
		On peut maintenant conclure : $S=\left\{\dfrac{\pi}{6};\dfrac{5\pi}{6}\right\}$	
	\end{corrBox}
	\item \ding{46} $\cos (x)=\dfrac{\sqrt{2}}{2}$ et $I=\left[-\dfrac{\pi}{2} ; \dfrac{\pi}{2}\right]$.
	\begin{corrBox}
		On commence par étudier la solution dans l'intervalle\\ $]-\pi;\pi]$ (parmi les mesures principales de l'angle).\\
		L'équation $\cos (x)=\dfrac{\sqrt{2}}{2}$ a pour solution :\\
		$\begin{cases}
			x_1 &=-\dfrac{\pi}{4}\\
			x_2 &=\dfrac{\pi}{4}
		\end{cases}$\\
		$x_1$ et $x_2$ sont les solutions dans l'intervalle $\left[-\dfrac{\pi}{2} ; \dfrac{\pi}{2}\right]$\\
		On peut maintenant conclure : $S=\left\{-\dfrac{\pi}{4};\dfrac{\pi}{4}\right\}$	
	\end{corrBox}
	\item $\sin (x)=-\dfrac{\sqrt{3}}{2}$ et $I=\left[-\pi ; \pi\right]$.
	\begin{corrBox}
		On commence par étudier la solution dans l'intervalle\\ $]-\pi;\pi]$ (parmi les mesures principales de l'angle).\\
		L'équation $\sin (x)=-\dfrac{\sqrt{3}}{2}$ a pour solution :\\
		$\begin{cases}
			x_1 &=-\dfrac{2\pi}{3}\\
			x_2 &=-\dfrac{\pi}{3}
		\end{cases}$\\
		$x_1$ et $x_2$ sont les solutions dans l'intervalle $\left[-\pi ; \pi\right]$\\
		On peut maintenant conclure : $S=\left\{-\dfrac{2\pi}{3};-\dfrac{\pi}{3}\right\}$	
	\end{corrBox}
	\item $\cos (x)=-\dfrac{\sqrt{2}}{2}$ et $I=\R$.
	\begin{corrBox}
		On commence par étudier la solution dans l'intervalle\\ $]-\pi;\pi]$ (parmi les mesures principales de l'angle).\\
		L'équation $\cos (x)=\dfrac{-\sqrt{2}}{2}$ a pour solution :\\
		$\begin{cases}
			x_1 &=-\dfrac{3\pi}{4}\\
			x_2 &=\dfrac{3\pi}{4}
		\end{cases}$\\
		$x_1$ et $x_2$ sont les solutions dans l'intervalle $]-\pi;\pi]$. On rajoute $2k\pi$ pour être dans \R\\
		On peut maintenant conclure : $S=\left\{-\dfrac{3\pi}{4}+2k\pi;\dfrac{3\pi}{4}+2k\pi\right\}$, $k\in\Z$	
	\end{corrBox}
	\item $\sin (x)=\dfrac{\sqrt{2}}{2}$ et $I=[\pi ; 3\pi]$.
	\begin{corrBox}
		On commence par étudier la solution dans l'intervalle\\ $]-\pi;\pi]$ (parmi les mesures principales de l'angle).\\
		L'équation $\sin (x)=\dfrac{\sqrt{2}}{2}$ a pour solution :\\
		$\begin{cases}
			x_1 &=\dfrac{\pi}{4}\\
			x_2 &=\dfrac{3\pi}{4}
		\end{cases}$\\
		$x_1$ et $x_2$ sont les solutions dans l'intervalle $]-\pi;\pi]$. On rajoute $2\pi$ pour être dans le bon intervalle\\
		On peut maintenant conclure : $S=\left\{\dfrac{9\pi}{4};\dfrac{11\pi}{4}\right\}$	
	\end{corrBox}
	\item $\cos (x)=-2$ et $I=\R$.
	\begin{corrBox}
		$-1\leqslant\cos(x)\leqslant1$ donc $S=\varnothing$	
	\end{corrBox}
	
\end{enumMet}
