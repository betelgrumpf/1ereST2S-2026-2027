\exerciceDS { - résolution d'équations - ( 7 points )}
Résoudre dans $\R$  les équations suivantes.
%\begin{multicols}{2}
    \begin{enumMet}
        \item $3 x^{2}+5 x-2 = 0$\quad \bareme{1}\\
        \begin{corrBox}
            $\Delta=b^2-4ac=5^{2}-4 \times 3 \times(-2)=49$ \\
            $\Delta>0$ donc l'équation admet deux solutions réelles.\\ 
            $x_{1}=\dfrac{-b-\sqrt{\Delta}}{2a}=\dfrac{-5-7}{6}=-2$ et $x_{2}=\dfrac{-b+\sqrt{\Delta}}{2a}=\dfrac{-5+7}{6}=\dfrac{1}{3}$ \\
            $S=\left\{-2;\dfrac{1}{3}\right\}$
        \end{corrBox}
        \item $3 x^{2}-x+5=0$\quad \bareme{1}\\
        \begin{corrBox}
                $\Delta=1^{2}-4 \times 3 \times 5=-59$ \\
                $\Delta<0$ donc aucune racine réelle.\\
                $S=\varnothing$
        \end{corrBox}
        \item $-2 x^{2}+7 x+4=0$\quad \bareme{1}\\
        \begin{corrBox}
            $\Delta=7^{2}-4 \times(-2) \times 4=81,$ \\
            $\Delta>0$ donc l'équation admet deux solutions réelles:\\
            $x_{1}=4$ et $x_{2}=-\dfrac{1}{2}$\\
            $S=\left\{-\dfrac{1}{2};4\right\}$
        \end{corrBox}
        \item $-x^{2}+x-7 = 0$\quad \bareme{1}\\
        \begin{corrBox}
            $\Delta=1^{2}-4 \times (-1) \times(-7)=-27$\\
            $\Delta<0$ donc aucune racine réelle.\\
            $S=\varnothing$
        \end{corrBox}
        \item $16 x^{2}-40 x+25 = 0$\quad \bareme{1,5}\\
        \begin{corrBox}
            On reconnaît une identité remarquable : $16 x^{2}-40 x+25=(4 x-5)^{2}$.\\
            Donc $16 x^{2}-40 x+25 = 0 \Leftrightarrow x=\dfrac{5}{4}$\\
            $S=\left\{\dfrac{5}{4}\right\}$
        \end{corrBox}
        \item $-3 x^{2}+5 x+8=0$\quad \bareme{1,5}\\
        \begin{corrBox}
            $\Delta=5^{2}-4 \times(-3) \times 8=121$\\
            $\Delta>0$ donc l'équation admet deux solutions réelles:\\ $x_{1}=\dfrac{8}{3}$ et $x_{2}=-1$\\
            $S=\left\{-1;\dfrac{8}{3}\right\}$
        \end{corrBox}
    \end{enumMet}
